/* ============================== USD/LKR EXCHANGE RATE ==============================
   NOTE: The Central Bank of Sri Lanka's own website does not expose a
   simple, stable, machine-readable rate feed — its published indicative
   rate is rendered by JavaScript inside the page, which a server-side
   script (this one) cannot execute. Confirmed by directly testing their
   site and the widget's backend endpoint; neither returns usable data
   from a plain HTTP request.
   As a practical alternative, this uses exchangerate-api.com's free,
   no-key "open" endpoint (open.er-api.com), which blends real market/
   bank-sourced rates and explicitly permits caching for commercial use.
   The fetched rate is cached in Script Properties for 12 hours so this
   only makes a live network call twice a day at most — everything else
   reads the cached value instantly. If the live fetch ever fails for
   any reason, this falls back to the last successfully fetched rate,
   or to a hardcoded value as an absolute last resort so property
   pricing never breaks due to an external service being unavailable. */

const USD_TO_LKR_FALLBACK_RATE_ = 335; // last-resort value only, if a live rate has never been fetched successfully
const USD_TO_LKR_REFRESH_INTERVAL_MS_ = 12 * 60 * 60 * 1000; // refetch at most twice a day

function getUsdToLkrRate_() {
  const props = PropertiesService.getScriptProperties();
  const cachedRate = props.getProperty('USD_TO_LKR_RATE_CACHED');
  const cachedAt = props.getProperty('USD_TO_LKR_RATE_FETCHED_AT');
  const now = new Date().getTime();

  if (cachedRate && cachedAt && (now - Number(cachedAt)) < USD_TO_LKR_REFRESH_INTERVAL_MS_) {
    return parseFloat(cachedRate);
  }

  try {
    const resp = UrlFetchApp.fetch('https://open.er-api.com/v6/latest/USD', { muteHttpExceptions: true });
    if (resp.getResponseCode() !== 200) throw new Error('HTTP ' + resp.getResponseCode());
    const data = JSON.parse(resp.getContentText());
    const rate = data && data.rates && data.rates.LKR;
    if (!rate || typeof rate !== 'number') throw new Error('LKR rate missing from response');

    props.setProperty('USD_TO_LKR_RATE_CACHED', String(rate));
    props.setProperty('USD_TO_LKR_RATE_FETCHED_AT', String(now));
    return rate;
  } catch (e) {
    // Prefer a stale-but-real previously fetched rate over the hardcoded
    // fallback, since it's still more accurate than a long-outdated constant.
    if (cachedRate) return parseFloat(cachedRate);
    return USD_TO_LKR_FALLBACK_RATE_;
  }
}

/* Public — called from the client right before building a property
   PDF, to turn the raw, often run-on "Other Information" free text into
   a clean bullet list. This ONLY affects what gets printed in the PDF —
   the original text in Supabase and in the property detail modal is
   never read back, overwritten, or touched in any way; the client
   fetches this fresh, on demand, purely for that one PDF. If the Gemini
   key isn't set, the API call fails, times out, or returns anything
   unexpected, this silently falls back to the ORIGINAL text unchanged —
   a PDF download must never be blocked by an AI hiccup. No session
   token is required, matching the other PDF-support functions
   (getPropertyPhotosForPdf) which are equally available to Guest/Agent
   downloads, since "Other Information" is already shown to them
   unredacted in the app. */
function formatTextWithAI(rawText) {
  rawText = s_(rawText);
  if (!rawText) return rawText;

  const key = getGeminiKey_();
  if (!key) return rawText; // no key configured — just use the original text as-is

  // Reasonable cap so a single call can't run away in cost or latency —
  // "Other Information" fields in practice are one short paragraph.
  const inputText = rawText.length > 3000 ? rawText.slice(0, 3000) : rawText;

  const prompt = 'Rewrite the following real estate listing notes as a clean, ' +
    'concise bullet-point list for a printed property PDF. While rewriting, fix ' +
    'any spelling and grammar mistakes in ordinary words (e.g. "Furnishesd" -> ' +
    '"Furnished") so the final text reads correctly. Do NOT change or "correct" ' +
    'any proper nouns (building names, street/road names, place names, person ' +
    'names) even if they look unusual \u2014 leave those exactly as written, since ' +
    '"fixing" a real name could make it wrong. Keep every factual detail exactly ' +
    'as given (numbers, prices, routes, measurements) \u2014 do not invent, omit, or ' +
    'change any facts. Do not add a heading, commentary, or any markdown ' +
    'formatting (no **, no #, no numbering). Output ONLY plain bullet lines, ' +
    'each starting with "\u2022 ", nothing else.\n\nNotes:\n' + inputText;

  try {
    const url = 'https://generativelanguage.googleapis.com/v1beta/models/' + GEMINI_MODEL + ':generateContent';
    const resp = UrlFetchApp.fetch(url, {
      method: 'POST',
      contentType: 'application/json',
      headers: { 'x-goog-api-key': key },
      payload: JSON.stringify({
        contents: [{ parts: [{ text: prompt }] }],
        generationConfig: { temperature: 0.2, maxOutputTokens: 500 }
      }),
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() !== 200) return rawText;
    const data = JSON.parse(resp.getContentText());
    const formatted = data && data.candidates && data.candidates[0] &&
      data.candidates[0].content && data.candidates[0].content.parts &&
      data.candidates[0].content.parts[0] && data.candidates[0].content.parts[0].text;
    return s_(formatted) || rawText;
  } catch (e) {
    return rawText; // network error, quota exceeded, malformed response, etc. — original text is always the safe fallback
  }
}

function supabaseRequest_(method, table, query, body, extraHeaders) {
  const key = getSupabaseKey_();
  let url = SUPABASE_URL + '/rest/v1/' + encodeURIComponent(table);
  if (query && query.length) url += '?' + query.join('&');
  const headers = { 'apikey': key, 'Authorization': 'Bearer ' + key };
  if (extraHeaders) Object.keys(extraHeaders).forEach(function (k) { headers[k] = extraHeaders[k]; });
  const options = { method: method, headers: headers, muteHttpExceptions: true };
  if (body !== undefined) {
    options.contentType = 'application/json';
    options.payload = JSON.stringify(body);
  }
  const resp = UrlFetchApp.fetch(url, options);
  const code = resp.getResponseCode();
  const text = resp.getContentText();
  if (code >= 400) throw new Error('Supabase ' + method + ' ' + table + ' failed (' + code + '): ' + text);
  return text ? JSON.parse(text) : null;
}

function supabaseRequestsParallel_(requests) {
  const key = getSupabaseKey_();
  const fetchRequests = requests.map(function (r) {
    let url = SUPABASE_URL + '/rest/v1/' + encodeURIComponent(r.table);
    if (r.query && r.query.length) url += '?' + r.query.join('&');
    const headers = { 'apikey': key, 'Authorization': 'Bearer ' + key };
    if (r.extraHeaders) Object.keys(r.extraHeaders).forEach(function (k) { headers[k] = r.extraHeaders[k]; });
    const options = { url: url, method: r.method, headers: headers, muteHttpExceptions: true };
    if (r.body !== undefined) {
      options.contentType = 'application/json';
      options.payload = JSON.stringify(r.body);
    }
    return options;
  });
  const responses = UrlFetchApp.fetchAll(fetchRequests);
  responses.forEach(function (resp, i) {
    const code = resp.getResponseCode();
    if (code >= 400) {
      throw new Error('Supabase ' + requests[i].method + ' ' + requests[i].table + ' failed (' + code + '): ' + resp.getContentText());
    }
  });
  return responses.map(function (resp) {
    const text = resp.getContentText();
    return text ? JSON.parse(text) : null;
  });
}

function getPropertiesByRefNos_(refNos) {
  const uniqueRefs = Array.from(new Set(refNos.filter(Boolean).map(function (r) { return s_(r).toUpperCase(); })));
  const map = {};
  if (!uniqueRefs.length) return map;
  const chunkSize = 200;
  for (let i = 0; i < uniqueRefs.length; i += chunkSize) {
    const chunk = uniqueRefs.slice(i, i + chunkSize);
    const inList = chunk.map(function (r) { return encodeURIComponent(r); }).join(',');
    const rows = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
      encodeURIComponent('Ref No') + '=in.(' + inList + ')', 'select=*'
    ]) || [];
    rows.forEach(function (obj) {
      const refNo = s_(obj['Ref No']).toUpperCase();
      if (!refNo) return;
      const cells = PROPERTY_COLUMN_NAMES.map(function (colName) { return obj[colName]; });
      map[refNo] = rowToObject_({ rowIndex: refNo, cells: cells });
    });
  }
  return map;
}

/* ==============================================================================
 * PATCHED: getAllDataRows_()
 *
 * Serializes concurrent full-table pulls with a script lock and briefly
 * caches each page (10s TTL) via CacheService. This fixes a Supabase
 * "statement timeout" (57014) that occurred when several google.script.run
 * calls fired on page load and each independently re-downloaded the
 * entire properties table at nearly the same instant, saturating the
 * database connection under concurrent load.
 * ============================================================================== */
/* Lightweight helper: asks Supabase for the exact row count of a table
   via the 'Prefer: count=exact' response header, without pulling any
   actual row data. Used so getAllDataRows_ knows exactly how many pages
   to request, and can fetch them all in one parallel batch instead of
   one at a time. Returns null (rather than throwing) if the count can't
   be determined for any reason, so callers can fall back safely. */
function getSupabaseRowCount_(table) {
  try {
    const key = getSupabaseKey_();
    const url = SUPABASE_URL + '/rest/v1/' + encodeURIComponent(table) + '?select=' + encodeURIComponent('Ref No') + '&limit=1';
    const resp = UrlFetchApp.fetch(url, {
      method: 'GET',
      headers: { 'apikey': key, 'Authorization': 'Bearer ' + key, 'Prefer': 'count=exact' },
      muteHttpExceptions: true
    });
    if (resp.getResponseCode() >= 400) return null;
    const headers = resp.getHeaders();
    const contentRange = headers['Content-Range'] || headers['content-range'] || '';
    const match = String(contentRange).match(/\/(\d+)$/);
    return match ? parseInt(match[1], 10) : null;
  } catch (e) {
    return null;
  }
}

function getAllDataRows_() {
  const lock = LockService.getScriptLock();
  // Wait up to 30s to acquire — if another execution is already pulling
  // the table, we queue behind it rather than firing a duplicate burst.
  const gotLock = lock.tryLock(30000);
  if (!gotLock) {
    throw new Error('Server is busy loading property data — please try again in a moment.');
  }

  try {
    // NOTE: pages are kept relatively small (1000 rows) to stay light
    // on the database per individual request, but fetched in PARALLEL —
    // wall-clock time is roughly the time of the slowest single page,
    // not the sum of all of them. A prior version fetched sequentially
    // to reduce concurrent DB load, but logs showed that alone taking
    // ~26s for a full pull — nearly the entire lock window for just ONE
    // caller — while Postgres itself showed zero actual statement
    // timeouts under real usage. Parallel fetch directly addresses the
    // actual bottleneck (wall-clock time / lock contention) that was
    // actually occurring.
    const pageSize = 1000;
    const out = [];
    const totalCount = getSupabaseRowCount_(SUPABASE_PROPERTIES_TABLE);

    if (totalCount !== null) {
      const pageCount = Math.max(1, Math.ceil(totalCount / pageSize));
      const requests = [];
      for (let i = 0; i < pageCount; i++) {
        requests.push({
          method: 'GET', table: SUPABASE_PROPERTIES_TABLE,
          query: [
            'select=*',
            'order=' + encodeURIComponent('Ref No') + '.asc',
            'limit=' + pageSize,
            'offset=' + (i * pageSize)
          ]
        });
      }
      const pages = supabaseRequestsParallel_(requests);
      pages.forEach(function (page) {
        (page || []).forEach(function (obj) {
          const refNo = s_(obj['Ref No']);
          if (!refNo) return;
          const cells = PROPERTY_COLUMN_NAMES.map(function (colName) { return obj[colName]; });
          out.push({ rowIndex: refNo, cells: cells });
        });
      });
    } else {
      // Fallback: the count probe failed for some reason — fetch
      // sequentially instead (slower, but safe and always correct).
      let offset = 0;
      while (true) {
        const page = fetchPropertiesPageWithRetry_(pageSize, offset);
        if (!page || !page.length) break;
        page.forEach(function (obj) {
          const refNo = s_(obj['Ref No']);
          if (!refNo) return;
          const cells = PROPERTY_COLUMN_NAMES.map(function (colName) { return obj[colName]; });
          out.push({ rowIndex: refNo, cells: cells });
        });
        if (page.length < pageSize) break;
        offset += pageSize;
      }
    }

    return out;
  } finally {
    lock.releaseLock();
  }
}

/* Fetches one page of the properties table, retrying once after a brief
   pause if the first attempt fails. Only used as a fallback if the row
   count probe above fails for some reason. */
function fetchPropertiesPageWithRetry_(pageSize, offset) {
  const query = [
    'select=*',
    'order=' + encodeURIComponent('Ref No') + '.asc',
    'limit=' + pageSize,
    'offset=' + offset
  ];
  try {
    return supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, query);
  } catch (e) {
    Utilities.sleep(1500);
    return supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, query);
  }
}

/* ============================== SESSIONS (server-side identity) ==============================
   A privileged call now proves identity with an opaque session token instead
   of a client-supplied name. Tokens are minted only inside verifyLoginPassword
   (after the password check succeeds) and stored SERVER-SIDE in CacheService,
   which the browser cannot read or write directly. Every function that used
   to trust a `currentUser` string parameter now takes `sessionToken` and
   calls resolveSession_/requireAdmin_/requireSocialMediaQueueAccess_ to
   resolve the real, verified identity before doing anything sensitive. */

const SESSION_TTL_SECONDS = 6 * 60 * 60; // CacheService max is 6 hours

function sessionCacheKey_(token) {
  return 'session_' + token;
}

/* Creates a new session for `name`, invalidating nothing else (multiple
   concurrent sessions for the same user are fine — e.g. phone + laptop). */
function createSession_(name, role) {
  const token = Utilities.getUuid();
  CacheService.getScriptCache().put(
    sessionCacheKey_(token),
    JSON.stringify({ name: name, role: role, issued: new Date().getTime() }),
    SESSION_TTL_SECONDS
  );
  return token;
}

/* Resolves a session token into { name, role } — or throws if the token is
   missing, unknown, or expired. This is the ONLY source of truth for "who
   is calling this function" anywhere sensitive in this file. Never trust
   a name/role passed in directly from the client instead of this. */
function resolveSession_(sessionToken) {
  const token = s_(sessionToken);
  if (!token) throw new Error('Not logged in — please log in again.');
  const raw = CacheService.getScriptCache().get(sessionCacheKey_(token));
  if (!raw) throw new Error('Your session has expired — please log in again.');
  const session = JSON.parse(raw);
  // Agent accounts (third-party self-service sign-ups, see
  // submitAgentSignup/verifyAgentLogin) are restricted to Search &
  // Filter only. Every one of those read-only functions
  // (searchProperties, getPropertyDetails, getPropertyPhotos,
  // generatePropertyPdf, etc.) takes no sessionToken at all — so ANY
  // function that requires a session is, by definition, staff-only.
  // Rejecting Agent sessions here, once, means every existing and
  // future requireLogin_/requireAdmin_/resolveSession_ caller is
  // automatically protected without having to individually re-check
  // the role everywhere.
  if (session.role === 'Agent') {
    throw new Error('This action is not available for Agent accounts.');
  }
  return session;
}

/* Sliding expiry: called by any authenticated action so a session doesn't
   die mid-use; re-writes the same token with a fresh TTL. */
function touchSession_(sessionToken, session) {
  CacheService.getScriptCache().put(
    sessionCacheKey_(sessionToken),
    JSON.stringify(session),
    SESSION_TTL_SECONDS
  );
}

/* Resolves the session and requires Admin role. Returns the verified name
   so callers can use it for attribution (Activity Log "User" column, etc.)
   without ever trusting a client-supplied name again. */
function requireAdmin_(sessionToken) {
  const session = resolveSession_(sessionToken);
  if (session.role !== 'Admin') {
    throw new Error('Only Admins (Keerthie, Sherden) can do this.');
  }
  touchSession_(sessionToken, session);
  return session.name;
}

/* Resolves the session for any logged-in (non-Guest) user and returns the
   verified name — used by actions any authenticated user can take. */
function requireLogin_(sessionToken) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  return session.name;
}

function logoutSession(sessionToken) {
  const token = s_(sessionToken);
  if (token) CacheService.getScriptCache().remove(sessionCacheKey_(token));
  return { success: true };
}

/* Used only by the client's "resume my session after a page refresh"
   logic (see attemptResume_ in JavaScript.html). The client stores the
   token in sessionStorage (survives a refresh, cleared when the tab
   closes) rather than localStorage, so this only ever fires on an
   actual refresh within the same tab, never on a genuinely fresh visit.
   This is a DELIBERATE, narrow exception to the rule in resolveSession_
   that rejects Agent-role tokens: this function only ever answers "is
   this token still valid, and if so who does it belong to" — it grants
   no additional capability an Agent's token didn't already carry, it
   just lets the browser silently restore the UI the person already had
   before refreshing. Every other authenticated function still goes
   through resolveSession_ and still rejects Agent tokens as before. */
function resumeSession(sessionToken) {
  const token = s_(sessionToken);
  if (!token) throw new Error('No session to resume.');
  const raw = CacheService.getScriptCache().get(sessionCacheKey_(token));
  if (!raw) throw new Error('Session expired — please log in again.');
  const session = JSON.parse(raw);
  touchSession_(token, session);
  return { name: session.name, role: session.role };
}

/* ============================== PASSWORD HASHING ============================== */

/* Salted SHA-256. Not as strong as bcrypt/scrypt (Apps Script has no
   built-in slow hash), but it is a large improvement over storing
   plaintext passwords, which is what this table did before. Stored format
   in the Password column is "salt:hexhash". */
function hashPassword_(password, salt) {
  const bytes = Utilities.computeDigest(Utilities.DigestAlgorithm.SHA_256, salt + password);
  return bytes.map(function (b) { return ((b < 0 ? b + 256 : b).toString(16)).padStart(2, '0'); }).join('');
}

function makePasswordRecord_(plainPassword) {
  const salt = Utilities.getUuid();
  return salt + ':' + hashPassword_(plainPassword, salt);
}

/* Verifies a plaintext password against a stored record. Transparently
   accepts a legacy plaintext record too (no ":" separator) so existing
   accounts keep working until migrateAllPasswordsToHashed() is run —
   remove the legacy branch once you've confirmed the migration ran. */
function verifyPasswordRecord_(plainPassword, stored) {
  stored = s_(stored);
  if (!stored) return false;
  const parts = stored.split(':');
  if (parts.length === 2) {
    const salt = parts[0], hash = parts[1];
    return hashPassword_(plainPassword, salt) === hash;
  }
  // Legacy plaintext fallback — only reached for accounts not yet migrated.
  return stored === plainPassword;
}

/* One-time migration — run this once from the Apps Script editor (function
   dropdown -> migrateAllPasswordsToHashed -> Run) after deploying this
   version. Converts every plaintext password still in Supabase into a
   salted hash. Already-hashed records (containing ":") are left alone, so
   this is safe to re-run. */
function migrateAllPasswordsToHashed() {
  const rows = supabaseRequest_('GET', SUPABASE_USERS_TABLE, ['select=id,Password']) || [];
  let migrated = 0;
  rows.forEach(function (r) {
    const current = s_(r['Password']);
    if (!current || current.indexOf(':') !== -1) return; // blank or already hashed
    const record = makePasswordRecord_(current);
    supabaseRequest_('PATCH', SUPABASE_USERS_TABLE, ['id=eq.' + encodeURIComponent(r.id)],
      { 'Password': record }, { 'Prefer': 'return=minimal' });
    migrated++;
  });
  return 'Migrated ' + migrated + ' password(s) to salted hashes.';
}

/* ============================== LOGIN RATE LIMITING ============================== */

const LOGIN_MAX_ATTEMPTS = 5;
const LOGIN_LOCKOUT_SECONDS = 300; // 5 minutes

function loginAttemptsKey_(name) {
  return 'loginattempts_' + s_(name).toLowerCase();
}

function checkLoginRateLimit_(name) {
  const cache = CacheService.getScriptCache();
  const key = loginAttemptsKey_(name);
  const attempts = Number(cache.get(key) || 0);
  if (attempts >= LOGIN_MAX_ATTEMPTS) {
    throw new Error('Too many failed attempts for this account. Try again in a few minutes.');
  }
}

function recordFailedLogin_(name) {
  const cache = CacheService.getScriptCache();
  const key = loginAttemptsKey_(name);
  const attempts = Number(cache.get(key) || 0) + 1;
  cache.put(key, String(attempts), LOGIN_LOCKOUT_SECONDS);
}

function clearLoginAttempts_(name) {
  CacheService.getScriptCache().remove(loginAttemptsKey_(name));
}

function s_(v) { return (v === null || v === undefined) ? '' : String(v).trim(); }

function levenshtein_(a, b) {
  const m = a.length, n = b.length;
  const d = [];
  for (let i = 0; i <= m; i++) d[i] = [i];
  for (let j = 0; j <= n; j++) d[0][j] = j;
  for (let i = 1; i <= m; i++) {
    for (let j = 1; j <= n; j++) {
      d[i][j] = a[i - 1] === b[j - 1] ? d[i - 1][j - 1] : 1 + Math.min(d[i - 1][j], d[i][j - 1], d[i - 1][j - 1]);
    }
  }
  return d[m][n];
}

function normalizeToClosest_(raw, canonicalList) {
  raw = s_(raw);
  if (!raw) return raw;
  if (canonicalList.indexOf(raw) !== -1) return raw;
  const rawLower = raw.toLowerCase();
  let best = null, bestDist = Infinity;
  canonicalList.forEach(function (opt) {
    const dist = levenshtein_(rawLower, opt.toLowerCase());
    if (dist < bestDist) { bestDist = dist; best = opt; }
  });
  return (best && bestDist <= Math.max(2, Math.round(best.length * 0.2))) ? best : raw;
}

function formatContactNumber_(raw) {
  raw = s_(raw);
  if (!raw) return raw;
  if (raw.charAt(0) === '0') return raw;
  if (raw.charAt(0) === '+') return '00' + raw.slice(1);
  const digitsOnly = raw.replace(/\D/g, '');
  if (!digitsOnly) return raw;
  return (digitsOnly.length === 9 ? '0' : '00') + raw;
}

function n_(v) {
  if (v === null || v === undefined || v === '') return null;
  const num = parseFloat(String(v).toString().replace(/[^0-9.\-]/g, ''));
  return isNaN(num) ? null : num;
}

function isoDate_(v) {
  if (!v) return '';
  let d = new Date(v);
  if (isNaN(d.getTime())) {
    const m = String(v).trim().match(/^(\d{1,2})[\/\-](\d{1,2})[\/\-](\d{4})(?:[ T](\d{1,2}):(\d{2})(?::(\d{2}))?)?/);
    if (m) {
      d = new Date(+m[3], +m[2] - 1, +m[1], +(m[4] || 0), +(m[5] || 0), +(m[6] || 0));
    }
  }
  return isNaN(d.getTime()) ? '' : d.toISOString();
}

/* ============================== AGENT SELF-SERVICE ACCOUNTS ============================== */
/* Third-party "Agent" accounts replace the old device-remembered Guest
   flow: an outside company fills in the same company-info form as
   before, and their registered mobile number automatically becomes
   BOTH their username and password (see submitAgentSignup) — nothing
   extra to create or remember. New signups start life as "Pending" and
   cannot log in until an Admin approves them (adminApproveAgent).
   Approved accounts can still be temporarily switched off via "Active"
   without losing their approval history (adminSetAgentActive), and are
   always restricted to Search & Filter with owner contact details
   hidden — enforced both in the client (isGuest checks in
   JavaScript.html now also cover role 'Agent') and, more importantly,
   on the server: resolveSession_ rejects an Agent's token for every
   function that requires a session, since none of the read-only search
   functions an Agent actually needs require one.

   SECURITY NOTE: because username and password are the same value (a
   business contact number that companies often share freely), this is
   effectively single-factor, guessable-if-known-publicly access. That
   trade-off is intentional here — Agent accounts only ever reach
   redacted, read-only search results, never owner contact details or
   any write action — but it should not be extended to any tier with
   broader access without adding a real, separate secret. */

function submitAgentSignup(payload) {
  payload = payload || {};

  // Honeypot: as_website is an input hidden from real visitors via CSS
  // in Index.html. A human never fills it in; a bot filling in every
  // field on the page will. If it's non-empty, silently report success
  // without writing anything, so a scripted submitter never learns the
  // honeypot tripped.
  if (s_(payload.website)) {
    return { success: true };
  }

  const company = s_(payload.company);
  const contactPerson = s_(payload.contactPerson);
  const email = s_(payload.email);
  const address = s_(payload.address);
  const consent = !!payload.consent;

  if (!company || !contactPerson || !s_(payload.contactNumber) || !email || !address) {
    throw new Error('Please fill all required fields.');
  }
  if (!/^\S+@\S+\.\S+$/.test(email)) {
    throw new Error('Please enter a valid email address.');
  }
  if (!consent) {
    throw new Error('Please confirm you agree to act as a partner for Central7.');
  }

  // The registered mobile number becomes BOTH the login username and
  // the password — normalized the same way Contact No 1/2 are
  // normalized everywhere else in this file, so a number entered
  // slightly differently at signup vs. login (e.g. with or without a
  // leading 0) still matches consistently.
  const mobileNumber = formatContactNumber_(s_(payload.contactNumber));

  const existing = supabaseRequest_('GET', SUPABASE_AGENTS_TABLE, [
    'Username=ilike.' + encodeURIComponent(mobileNumber), 'select=id', 'limit=1'
  ]);
  if (existing && existing.length) {
    throw new Error('This mobile number is already registered. Contact Central7 if you need help logging in.');
  }

  supabaseRequest_('POST', SUPABASE_AGENTS_TABLE, null,
    [{
      'Company Name': company,
      'Contact Person': contactPerson,
      'Contact Number': mobileNumber,
      'Email': email,
      'Registered Address': address,
      'Username': mobileNumber,
      'Password': makePasswordRecord_(mobileNumber),
      'Status': 'Pending',
      'Active': 'true',
      'Created At': new Date()
    }],
    { 'Prefer': 'return=minimal' });

  return { success: true };
}

/* Called from the Agent Log In form — a single "registered mobile
   number" field, since that value is both the username and the
   password (see submitAgentSignup). Rate-limited per number (using a
   distinct 'agent:' cache-key prefix so it never collides with staff
   login attempt counters), and only issues a session if the account
   has been Approved by an Admin and is still Active. */
function verifyAgentLogin(mobileNumber) {
  mobileNumber = formatContactNumber_(s_(mobileNumber));
  if (!mobileNumber) throw new Error('Please enter your registered mobile number.');
  const rateLimitKey = 'agent:' + mobileNumber.toLowerCase();
  checkLoginRateLimit_(rateLimitKey);

  const rows = supabaseRequest_('GET', SUPABASE_AGENTS_TABLE, [
    'Username=ilike.' + encodeURIComponent(mobileNumber), 'select=*', 'limit=1'
  ]);
  const agent = rows && rows[0];
  if (!agent || !verifyPasswordRecord_(mobileNumber, agent['Password'])) {
    recordFailedLogin_(rateLimitKey);
    throw new Error('No account found for that mobile number.');
  }

  const status = s_(agent['Status']);
  if (status === 'Pending') {
    throw new Error('Your account is still awaiting Admin approval.');
  }
  if (status === 'Rejected') {
    throw new Error('Your account request was not approved. Contact Central7 for details.');
  }
  if (status !== 'Approved' || s_(agent['Active']) !== 'true') {

  const token = createSession_(displayName, 'Agent');
  return { success: true, token: token, role: 'Agent', name: displayName };
}

/* ============================== DROPDOWN DATA ============================== */

function getFormOptions() {
  const citiesSet = {};
  let offset = 0;
  const pageSize = 5000;
  while (true) {
    const page = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
      'select=' + encodeURIComponent('City'), 'limit=' + pageSize, 'offset=' + offset
    ]);
    if (!page || !page.length) break;
    page.forEach(function (obj) { const city = s_(obj['City']); if (city) citiesSet[city] = true; });
    if (page.length < pageSize) break;
    offset += pageSize;
  }
  const cities = Object.keys(citiesSet).sort();
  const allUsers = listUsers_();
  return {
    users: allUsers.filter(function (u) { return u.active; }).map(function (u) { return u.name; }),
    usersDetailed: allUsers,
    opportunityTypes: OPPORTUNITY_TYPES,
    propertyTypes: PROPERTY_TYPES,
    statusList: STATUS_LIST,
    furnishedList: FURNISHED_LIST,
    contactTypes: CONTACT_TYPES,
    amenities: AMENITIES_LIST,
    apartmentComplexes: getApartmentComplexes(),
    cities: cities
  };
}

function listUsers_() {
  const rows = supabaseRequest_('GET', SUPABASE_USERS_TABLE, ['select=*', 'order=Name.asc']);
  return (rows || [])
    .map(function (r) {
      const photoId = s_(r['Photo']);

        mobile: s_(r['Mobile Number']),
        photoUrl: photoId ? ('https://drive.google.com/thumbnail?id=' + photoId + '&sz=w200') : ''
      };
      // Password is intentionally never included here.
    })
    .filter(function (u) { return u.name; })

}

function getUserPassword_(name) {
  const users = listUsers_();
  const target = users.find(function (u) { return u.name.toLowerCase() === s_(name).toLowerCase(); });
  if (!target) return '';
  const rows = supabaseRequest_('GET', SUPABASE_USERS_TABLE, ['select=Password', 'id=eq.' + encodeURIComponent(target.id)]);
  return (rows && rows[0]) ? s_(rows[0]['Password']) : '';
}

/* Called from the login screen after picking a name + entering a password.
   Rate-limited per account name, verifies against the salted hash (with a
   transparent legacy-plaintext fallback until migrateAllPasswordsToHashed
   has been run), and — on success — mints and returns a server-side
   session token. The client stores ONLY this opaque token from now on;
   it is never allowed to just assert "I am X" for anything sensitive. */
function verifyLoginPassword(name, password) {
  name = s_(name);
  if (!name) throw new Error('Please select your name.');
  checkLoginRateLimit_(name);

  const stored = getUserPassword_(name);
  if (!stored) {
    recordFailedLogin_(name);
    throw new Error('No password set for this user yet — ask an Admin to set one in the Users tab.');
  }
  if (!verifyPasswordRecord_(password, stored)) {
    recordFailedLogin_(name);
    throw new Error('Incorrect password.');
  }

  clearLoginAttempts_(name);
  const role = getUserRole_(name);
  const token = createSession_(name, role);
  return { success: true, token: token, role: role };
}

function generateRandomPassword_() {
  const chars = 'ABCDEFGHJKLMNPQRSTUVWXYZabcdefghjkmnpqrstuvwxyz23456789!@#$%';
  let pwd = '';
  for (let i = 0; i < 10; i++) {
    pwd += chars.charAt(Math.floor(Math.random() * chars.length));
  }
  return pwd;
}

/* One-time setup — now stores salted hashes rather than plaintext. Safe to
   re-run only if you actually want to reset everyone back to these values. */

function userSortComparator_(a, b) {
  const ia = USER_DISPLAY_ORDER.indexOf(a.name);
  const ib = USER_DISPLAY_ORDER.indexOf(b.name);
  const ra = ia === -1 ? USER_DISPLAY_ORDER.length : ia;
  const rb = ib === -1 ? USER_DISPLAY_ORDER.length : ib;
  if (ra !== rb) return ra - rb;
  return a.name.localeCompare(b.name);
}

/* ============================== SEARCH & FILTER (public — no login required) ============================== */

function searchProperties(filters) {
  filters = filters || {};
  const rows = getAllDataRows_().map(rowToObject_);

  const wantMinPriceLKR = filters.priceMin ? (filters.priceMin * (filters.priceMinIsUSD ? USD_TO_LKR_RATE : 1)) : null;
  const wantMaxPriceLKR = filters.priceMax ? (filters.priceMax * (filters.priceMaxIsUSD ? USD_TO_LKR_RATE : 1)) : null;

  const results = rows.filter(function (p) {
    if (filters.opportunityType && filters.opportunityType !== 'All' && p.opportunityType !== filters.opportunityType) return false;
    if (filters.propertyTypes && filters.propertyTypes.length) {
      if (filters.propertyTypes.indexOf(p.propertyType) === -1) return false;
    } else if (filters.propertyType && filters.propertyType !== 'All' && p.propertyType !== filters.propertyType) return false;
    if (filters.status && filters.status !== 'All' && p.status !== filters.status) return false;
    if (filters.agent && filters.agent !== 'All' && p.user !== filters.agent) return false;
    if (filters.furnished && filters.furnished !== 'All' && p.furnished !== filters.furnished) return false;
    if (filters.cities && filters.cities.length && filters.cities.indexOf(p.city) === -1) return false;
    if (filters.refNoContains && p.refNo.toLowerCase().indexOf(filters.refNoContains.toLowerCase()) === -1) return false;
    if (filters.nameContains && p.name.toLowerCase().indexOf(filters.nameContains.toLowerCase()) === -1) return false;
    if (filters.phoneContains) {
      const digits = filters.phoneContains.replace(/\D/g, '').replace(/^0+/, '');
      const combined = (p.contact1 + p.contact2).replace(/\D/g, '').replace(/^0+/, '');
      if (digits && combined.indexOf(digits) === -1) return false;
    }
    if (filters.minBedrooms && (!p.bedRooms || p.bedRooms < filters.minBedrooms)) return false;
    if (filters.keyword) {
      const kw = filters.keyword.toLowerCase();
      const hay = (p.address + ' ' + p.purpose + ' ' + p.otherInfo + ' ' + p.amenities).toLowerCase();
      if (hay.indexOf(kw) === -1) return false;
    }
    if (filters.floorAreaMin && (!p.floorArea || p.floorArea < filters.floorAreaMin)) return false;
    if (filters.floorAreaMax && (!p.floorArea || p.floorArea > filters.floorAreaMax)) return false;
    if (filters.landSizeMin && (!p.landSize || p.landSize < filters.landSizeMin)) return false;
    if (filters.landSizeMax && (!p.landSize || p.landSize > filters.landSizeMax)) return false;
    if (wantMinPriceLKR && (p.priceInLKR === null || p.priceInLKR < wantMinPriceLKR)) return false;
    if (wantMaxPriceLKR && (p.priceInLKR === null || p.priceInLKR > wantMaxPriceLKR)) return false;
    if (filters.amenities && filters.amenities.length) {
      const amenLower = p.amenities.toLowerCase();
      const missing = filters.amenities.some(function (a) { return amenLower.indexOf(a.toLowerCase()) === -1; });
      if (missing) return false;
    }
    return true;
  });

  results.sort(function (a, b) { return (b.timestamp || '').localeCompare(a.timestamp || ''); });
  return filters.noLimit ? results : results.slice(0, 500);
}

/* Regular users only ever see their own properties — targetUser lets an
   Admin look at someone else's. Identity + role now come from the
   session, not from the client-supplied currentUser string. */
function getMyProperties(sessionToken, targetUser) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  const user = targetUser || session.name;
  if (user.toLowerCase() !== session.name.toLowerCase() && session.role !== 'Admin') {
    throw new Error('Only an Admin can view another user\'s properties.');
  }
  // noLimit: true — "My Properties" must show every matching property,
  // not just the 500 most recent. The general Search & Filter tab
  // (searchProperties without this flag) keeps that cap since a broad
  // search there could otherwise return thousands of rows, but My
  // Properties is specifically for looking up a known/older property —
  // capping it silently hid anything not among the newest 500 out of
  // 10,000+ total records (e.g. an older listing like C7-8718 simply
  // never appeared, even when searched for directly by Ref No).
  return searchProperties({ agent: user, status: 'All', noLimit: true });
}

/* ============================== PARSE A CUSTOMER INQUIRY (public) ============================== */

function parseInquiryText(text) {
  text = text || '';
  const lower = text.toLowerCase().replace(/[\u2013\u2014\u2212]/g, '-');
  const filters = { status: 'Active' };

  if (/\brent(ing)?\s*wanted|\blooking to rent\b|\bwant(s|ed)? to rent\b|\brent\b/.test(lower)) filters.opportunityType = 'Rent Out';
  else filters.opportunityType = 'Sell';

  if (/\bindustrial\b|\bwarehouse\b|\bfactory\b/.test(lower)) filters.propertyTypes = ['Land', 'Commercial Property'];
  else if (/\bapartment\b|\bflat\b/.test(lower)) filters.propertyType = 'Apartment';
  else if (/\bland\b|\bplot\b/.test(lower)) filters.propertyType = 'Land';
  else if (/\bcommercial\b|\bshop\b|\boffice\b/.test(lower)) filters.propertyType = 'Commercial Property';
  else if (/\bhouse\b/.test(lower)) filters.propertyType = 'House';

  const options = getFormOptions();

  const foundCities = [];
  const colomboListMatch = lower.match(/\bcol(?:ombo)?[\s:-]*((?:\d{1,2}\s*(?:,|&|\band\b|\bor\b|\/|-)?\s*)+)/);
  if (colomboListMatch) {
    const nums = colomboListMatch[1].match(/\d{1,2}/g) || [];
    nums.forEach(function (n) {
      const padded = 'Colombo ' + (n.length === 1 ? '0' + n : n);
      if (options.cities.indexOf(padded) !== -1 && foundCities.indexOf(padded) === -1) foundCities.push(padded);
    });
  }
  const CITY_ALIASES = {
    dehiwela: 'Dehiwala', dehiwala: 'Dehiwala',
    nugegoda: 'Nugegoda',
    'mount lavinia': 'Mount Lavinia', 'mt lavinia': 'Mount Lavinia',
    wellawatte: 'Colombo 06', wellawatta: 'Colombo 06',
    kolpity: 'Colombo 03', kollupitiya: 'Colombo 03', collupitiya: 'Colombo 03', collpitiya: 'Colombo 03', kolupity: 'Colombo 03',
    bamba: 'Colombo 04', bambalapitiya: 'Colombo 04'
  };
  Object.keys(CITY_ALIASES).forEach(function (alias) {
    const target = CITY_ALIASES[alias];
    if (lower.indexOf(alias) !== -1 && options.cities.indexOf(target) !== -1 && foundCities.indexOf(target) === -1) foundCities.push(target);
  });
  options.cities.forEach(function (city) {
    if (foundCities.indexOf(city) !== -1) return;
    if (city.toLowerCase().indexOf('colombo') !== -1) return;
    if (lower.indexOf(city.toLowerCase()) !== -1) foundCities.push(city);
  });
  if (foundCities.length) filters.cities = foundCities;

  const complexes = getApartmentComplexes();
  const matchedComplex = complexes.find(function (c) {
    if (!c.name) return false;
    const nameLower = c.name.toLowerCase();
    const escape = function (s) { return s.replace(/[.*+?^${}()|[\]\\]/g, '\\$&'); };
    if (new RegExp('\\b' + escape(nameLower) + '\\b').test(lower)) return true;
    const firstWord = nameLower.split(/\s+/)[0];
    return firstWord.length >= 3 && new RegExp('\\b' + escape(firstWord) + '\\b').test(lower);
  });
  if (matchedComplex) {
    if (!filters.cities || !filters.cities.length) {
      if (matchedComplex.location && options.cities.indexOf(matchedComplex.location) !== -1) {
        filters.cities = [matchedComplex.location];
      }
    }
  }

  const perchMatch = lower.match(/(\d+(\.\d+)?)\s*(-|to)?\s*(\d+(\.\d+)?)?\s*perch/);
  if (perchMatch) {
    if (perchMatch[4]) {
      // A real range was given (e.g. "6-15 perch") — use both bounds as specified.
      filters.landSizeMin = parseFloat(perchMatch[1]);
      filters.landSizeMax = parseFloat(perchMatch[4]);
    } else {
      // Single approximate value (e.g. "15 perch") — treat as a rough
      // target, not a strict minimum. Only cap the upper bound (+3), so
      // smaller properties that might still be a good fit aren't
      // invisibly excluded.
      filters.landSizeMax = parseFloat(perchMatch[1]) + 3;
    }
  } else {
    // No "perch" mentioned — check for acres instead (1 acre = 160 perch).
    // Handles both "5 acre - 15 acre" (unit repeated after each number)
    // and "5-15 acres" / "5 acres" (unit given once, at the end).
    let acreMatch = lower.match(/(\d+(?:\.\d+)?)\s*acres?\s*(?:-|to)\s*(\d+(?:\.\d+)?)\s*acres?/);
    if (!acreMatch) acreMatch = lower.match(/(\d+(?:\.\d+)?)\s*(?:-|to)?\s*(\d+(?:\.\d+)?)?\s*acres?/);
    if (acreMatch) {
      const ACRE_TO_PERCH = 160;
      const acreMin = parseFloat(acreMatch[1]);
      if (acreMatch[2]) {
        // Real range given — use both bounds as specified.
        filters.landSizeMin = acreMin * ACRE_TO_PERCH;
        filters.landSizeMax = parseFloat(acreMatch[2]) * ACRE_TO_PERCH;
      } else {
        // Single approximate value — only cap the upper bound.
        filters.landSizeMax = (acreMin + 0.5) * ACRE_TO_PERCH;
      }
    }
  }

  const bedMatch = lower.match(/(\d+)\s*(bed\s*rooms?|bedrooms?|beds?|rooms?|br\b)/);
  if (bedMatch) filters.minBedrooms = parseInt(bedMatch[1], 10);

  if (/\bunfurnished\b|\bnot\s+furnished\b|\bno\s+furnishing\b/.test(lower)) filters.furnished = 'Not Furnished';
  else if (/\bfully\s+furnished\b/.test(lower)) filters.furnished = 'Fully Furnished';
  else if (/\bpartly\s+furnished\b|\bpartially\s+furnished\b/.test(lower)) filters.furnished = 'Partly Furnished';
  else if (/\bfurnished\b/.test(lower)) filters.furnished = 'Fully Furnished';

  const isUSD = /\busd\b|\$/.test(lower);

  const budgetIdx = lower.indexOf('budget');
  const budgetSection = budgetIdx !== -1 ? lower.slice(budgetIdx, budgetIdx + 80) : lower;

  const rangeMatch = budgetSection.match(/(\d+(\.\d+)?)\s*(?:million\b|mil\b|mn\b|m\b)?\s*(?:-|to|and)\s*(\d+(\.\d+)?)\s*(million\b|mil\b|mn\b|m\b)(?!\s*(?:from|radius|away|wide|deep|long|high|of)\b)/);
  const rangeMatchThousand = budgetSection.match(/(\d+(\.\d+)?)\s*k?\s*(?:-|to|and)\s*(\d+(\.\d+)?)\s*k\b(?!\s*(?:from|radius|away|wide|deep|long|high|of)\b)/);
  const millionMatch = budgetSection.match(/(\d+(\.\d+)?)\s*(million\b|mil\b|mn\b|m\b)(?!\s*(?:from|radius|away|wide|deep|long|high|of)\b)(?!\s*(?:-|to|and)\s*\d+(?:\.\d+)?\s*(?:million\b|mil\b|mn\b|m\b)?\s*(?:from|radius|away|wide|deep|long|high|of)\b)/);
  const thousandMatch = budgetSection.match(/(\d+(\.\d+)?)\s*k\b(?!\s*(?:sq\.?\s*ft|sqft|square\s*feet|sq\.?\s*m|sqm|perch(?:es)?|acres?|cents?))(?!\s*(?:from|radius|away|wide|deep|long|high|of)\b)(?!\s*(?:-|to|and)\s*\d+(?:\.\d+)?\s*k?\b\s*(?:from|radius|away|wide|deep|long|high|of)\b)/);
  const plainBudgetMatch = budgetSection.match(/(?:lkr|rs\.?)?\s*([\d,]{6,})(?!\d)(?!\s*(?:sq\.?\s*ft|sqft|square\s*feet|sq\.?\s*m|sqm|perch(?:es)?|acres?|cents?)\b)(?!\s*(?:-|to)\s*[\d,]+(?:\.\d+)?\s*(?:sq\.?\s*ft|sqft|square\s*feet|sq\.?\s*m|sqm|perch(?:es)?|acres?|cents?)\b)/);

  if (rangeMatch) {
    filters.priceMin = parseFloat(rangeMatch[1]) * 1000000;
    filters.priceMax = parseFloat(rangeMatch[3]) * 1000000;
  } else if (rangeMatchThousand) {
    filters.priceMin = parseFloat(rangeMatchThousand[1]) * 1000;
    filters.priceMax = parseFloat(rangeMatchThousand[3]) * 1000;
  } else if (millionMatch) {
    filters.priceMax = parseFloat(millionMatch[1]) * 1000000;
  } else if (thousandMatch) {
    filters.priceMax = parseFloat(thousandMatch[1]) * 1000;
  } else if (plainBudgetMatch) {
    filters.priceMax = parseFloat(plainBudgetMatch[1].replace(/,/g, ''));
  }
  if (filters.priceMin) filters.priceMinIsUSD = isUSD;
  if (filters.priceMax) filters.priceMaxIsUSD = isUSD;

  const results = searchProperties(filters);
  return { filters: filters, results: results };
}

function generateNextRefNo() {
  const rows = getAllDataRows_();
  let max = 0;
  rows.forEach(function (r) {
    const ref = s_(r.cells[COL.REF_NO - 1]);
    const m = ref.match(/(\d+)$/);
    if (m) {
      const num = parseInt(m[1], 10);
      if (num > max) max = num;
    }
  });
  return REF_PREFIX + (max + 1);
}

   writing anything — previously any client could call this unauthenticated. */
function saveProperty(sessionToken, formObj) {
  requireLogin_(sessionToken);
  const lock = LockService.getScriptLock();
  lock.waitLock(15000);
  try {
    const refNo = generateNextRefNo();

    if (!formObj.name || !formObj.contact1 || !formObj.opportunityType || !formObj.propertyType || !formObj.city) {
      throw new Error('Name of Contact, Contact No 1, Opportunity Type, Property Type and City are required.');
    }

    if (!doNotPublish) {

      if (!requestedPlatforms.length) {
        throw new Error('Select at least one platform to request publishing on.');
      }
    }

    const row = new Array(TOTAL_COLS).fill('');
    row[COL.REF_NO - 1] = refNo;
    row[COL.TIMESTAMP - 1] = new Date();

    row[COL.OPP_TYPE - 1] = formObj.opportunityType || '';
    row[COL.PROP_TYPE - 1] = formObj.propertyType || '';
    row[COL.ADDRESS - 1] = formObj.address || '';
    row[COL.CITY - 1] = formObj.city || '';

    const amenitiesText = [formObj.amenities && formObj.amenities.length ? formObj.amenities.join(', ') : '', formObj.otherAmenitiesFreeText || '']
      .filter(Boolean).join(', ');

    if (formObj.propertyType === 'House' || formObj.propertyType === 'Estate') {
      row[COL.PURPOSE - 1] = formObj.purpose || '';
      row[COL.H_LAND_SIZE - 1] = formObj.landSize || '';
      row[COL.H_BEDROOMS - 1] = formObj.bedRooms || '';
      row[COL.H_BATHROOMS - 1] = formObj.bathRooms || '';
      row[COL.H_FLOOR_AREA - 1] = formObj.floorArea || '';
      row[COL.H_FLOORS - 1] = formObj.noOfFloors || '';
      row[COL.H_PARKING - 1] = formObj.parkingSpaces || '';
      row[COL.H_AGE - 1] = formObj.age || '';
      row[COL.H_AMENITIES - 1] = amenitiesText;
      row[COL.H_OTHER_INFO - 1] = formObj.comments || '';
    } else if (formObj.propertyType === 'Land') {
      row[COL.L_LAND_SIZE - 1] = formObj.landSize || '';
      row[COL.L_SUITABLE - 1] = formObj.suitableFor || '';
      row[COL.L_AMENITIES - 1] = amenitiesText;
      row[COL.L_OTHER_INFO - 1] = formObj.comments || '';
    } else if (formObj.propertyType === 'Apartment') {
      row[COL.A_COMPLEX - 1] = formObj.apartmentComplex || '';
      row[COL.A_FLOOR - 1] = formObj.floor || '';
      row[COL.A_ROOMS - 1] = formObj.bedRooms || '';
      row[COL.A_BATHROOMS - 1] = formObj.bathRooms || '';
      row[COL.A_FLOOR_AREA - 1] = formObj.floorArea || '';
      row[COL.A_PARKING - 1] = formObj.parkingSpaces || '';
      row[COL.A_VIEW - 1] = formObj.view || '';

      row[COL.C_BUILTUP - 1] = formObj.builtUpArea || formObj.floorArea || '';

    row[COL.PRICE_PER_SQFT - 1] = formObj.pricePerSqft || '';

    row[COL.BUDGET - 1] = formObj.budget || '';
    row[COL.COMMENTS - 1] = formObj.comments || '';
    row[COL.FURNISHED - 1] = formObj.furnished || '';
    row[COL.STATUS - 1] = formObj.status || 'Active';

    insertPropertyRow_(row);

    // "Do Not Publish" entries are record-only — the property still
    // gets saved and searchable like any other, but deliberately never
    // enters the Activity Log / Social Media Queue approval workflow at
    // all, since there's nothing to publish or approve.
    if (!doNotPublish) {
      // CHANGED: new-listing Activity Log entries are now routed exclusively
      // to Keerthie via "Assigned To", using the same mechanism as Boost

        }],
        { 'Prefer': 'return=minimal' });

    return { success: true, refNo: refNo };
  } finally {
    lock.releaseLock();
  }
}

function getStatusUpdateLog(sessionToken, dateFrom, dateTo, statusFilter) {
  const currentAdmin = requireAdmin_(sessionToken);
  const dtCol = encodeURIComponent('"Date & Time"');
  const query = ['select=*', 'order=' + dtCol + '.desc', 'limit=2000'];
  if (dateFrom) query.push(dtCol + '=gte.' + encodeURIComponent(dateFrom + 'T00:00:00.000Z'));
  if (dateTo) query.push(dtCol + '=lte.' + encodeURIComponent(dateTo + 'T23:59:59.999Z'));
  if (statusFilter === 'Publish, Republish, Data Change & Boost') {
    query.push('Status=in.(Publish,Republish,Data%20Change,Boost%20Request)');
  } else if (statusFilter && statusFilter !== 'All') {
    query.push('Status=eq.' + encodeURIComponent(statusFilter));
  }

  const results = supabaseRequestsParallel_([
    { method: 'GET', table: SUPABASE_STATUS_UPDATE_TABLE, query: query },
    { method: 'GET', table: SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, query: ['select=' + encodeURIComponent('Ref No,Approved Date,Completed Date')] }
  ]);
  const rows = results[0];
  const queueRows = results[1] || [];

  const lastQueuedAt = {};
  queueRows.forEach(function (q) {
    const ref = s_(q['Ref No']).toUpperCase();
    const approvedIso = isoDate_(q['Approved Date']);
    if (!approvedIso) return;
    if (!lastQueuedAt[ref] || approvedIso > lastQueuedAt[ref]) lastQueuedAt[ref] = approvedIso;
  });

  return (rows || [])
    .map(function (r) {
      return {
        dateTime: isoDate_(r['Date & Time']),
        refNo: s_(r['Ref No']),
        user: s_(r['User']),
        status: s_(r['Status']),
        comment: s_(r['Comment']),
        requestedPlatforms: parseRequestedPlatformsRaw_(r['Requested Platforms']),
        boostCompleted: !!s_(r['Boost Completed']),
        assignedTo: s_(r['Assigned To'])
      };
    })
    .filter(function (r) {
      if (r.status === 'Boost Request') {
        // Each admin only ever sees requests routed to them — Sherden
        // never sees Keerthie's Ikman bump-ups and vice versa.
        return !r.boostCompleted && r.assignedTo.toLowerCase() === currentAdmin.toLowerCase();
      }
      // CHANGED: any row that carries an "Assigned To" (e.g. new
      // listings / Republish, routed to Keerthie) is now only shown to
      // the admin it's assigned to. Rows with no "Assigned To" (Drop,
      // Lost, Hold, Closed, Data Change, etc.) remain visible to every
      // Admin, same as before.
      if (r.assignedTo && r.assignedTo.toLowerCase() !== currentAdmin.toLowerCase()) {
        return false;
      }
      const lastQueued = lastQueuedAt[r.refNo.toUpperCase()];
      return !lastQueued || r.dateTime > lastQueued;
    });
}

/* ============================== BOOST REQUESTS (Ikman/Facebook ad pump-ups) ==============================
   Any logged-in user (not just Admin) can request a property's existing
   ad be "boosted"/pumped up on Ikman and/or Facebook, from a card on
   their My Dashboard. This lands directly in the Activity Log as a
   'Boost Request' entry — but routed to ONE SPECIFIC admin based on
   platform (Ikman bump-ups go to Keerthie, Facebook boosts go to
   Sherden), and each admin's Activity Log only shows requests assigned
   to them (see getStatusUpdateLog above). This NEVER touches the Social
   Media Queue at all, since a boost isn't a new listing/removal action,
   just a paid visibility bump on an ad that's already live. Completion
   is tracked via a dedicated "Boost Completed" column on the Status
   Update row itself (the table has no id column, so Ref No + Date & Time
   + Assigned To together are used to target the exact row — if both
   platforms are requested at once, they share the same timestamp but
   land in two separate rows distinguished by Assigned To). */

const BOOST_PLATFORMS = ['Ikman', 'Facebook'];
const BOOST_PLATFORM_ADMIN_ROUTING = { 'Ikman': 'Keerthie', 'Facebook': 'Sherden' };

// New-listing (Publish) entries, and Republish entries (whether triggered
// from My Properties or the Republish Queue), are always routed to
// Keerthie only, using the same "Assigned To" column already used above
// for Boost Requests. Other status changes (Drop/Lost/Hold/Closed/Data
// Change) are left unassigned and stay visible to any Admin, as before.
// See getStatusUpdateLog() for the matching read-side filter.
const ACTIVITY_LOG_NEW_LISTING_ADMIN = 'Keerthie';

function submitBoostRequest(sessionToken, refNo, platforms, comment) {
  const currentUser = requireLogin_(sessionToken);
  refNo = s_(refNo).toUpperCase();
  if (!refNo) throw new Error('Ref No is required.');

  const validPlatforms = (platforms || []).filter(function (p) { return BOOST_PLATFORMS.indexOf(p) !== -1; });
  if (!validPlatforms.length) throw new Error('Select at least one platform (Ikman or Facebook) to boost.');

  const existing = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
    encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo), 'select=' + encodeURIComponent('Ref No'), 'limit=1'
  ]);
  if (!existing || !existing.length) throw new Error('Property ' + refNo + ' not found — check the Ref No.');

  const now = new Date();
  // One row per platform, each routed to that platform's responsible
  // admin, so Keerthie and Sherden can each independently see and
  // complete only their own part of the request.
  const rows = validPlatforms.map(function (plat) {
    return {
      'Date & Time': now, 'Ref No': refNo, 'User': currentUser, 'Status': 'Boost Request',
      'Comment': s_(comment), 'Requested Platforms': plat,
      'Assigned To': BOOST_PLATFORM_ADMIN_ROUTING[plat] || '', 'Boost Completed': null
    };
  });
  supabaseRequest_('POST', SUPABASE_STATUS_UPDATE_TABLE, null, rows, { 'Prefer': 'return=minimal' });

  return { success: true, refNo: refNo, platforms: validPlatforms };
}

/* Admin-only — marks a specific Boost Request row done, so it drops out
   of the Activity Log. Only the admin it's assigned to can complete it.
   Does not write anything to the Social Media Queue table at all. */
function completeBoostRequest(sessionToken, refNo, dateTime, assignedTo) {
  const currentAdmin = requireAdmin_(sessionToken);
  if (s_(assignedTo).toLowerCase() !== currentAdmin.toLowerCase()) {
    throw new Error('This request is assigned to ' + assignedTo + '.');
  }
  refNo = s_(refNo).toUpperCase();
  const dtCol = encodeURIComponent('"Date & Time"');
  supabaseRequest_('PATCH', SUPABASE_STATUS_UPDATE_TABLE, [
    encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo),
    dtCol + '=eq.' + encodeURIComponent(dateTime),
    'Assigned%20To=eq.' + encodeURIComponent(assignedTo)
  ], { 'Boost Completed': new Date() }, { 'Prefer': 'return=minimal' });
  return { success: true };
}

function getRepublishQueue(sessionToken, agentFilter) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  const isAdmin = session.role === 'Admin';
  const effectiveAgent = isAdmin ? (agentFilter || 'All') : session.name;

  const queueRows = supabaseRequest_('GET', SUPABASE_REPUBLISH_QUEUE_TABLE, ['select=*', 'order=id.asc']) || [];
  const filteredQueue = queueRows
    .filter(function (q) { return effectiveAgent === 'All' || s_(q['User Name']) === effectiveAgent; })
    .filter(function (q) { return !s_(q['Status']); });
  if (!filteredQueue.length) return [];

  const propsByRef = getPropertiesByRefNos_(filteredQueue.map(function (q) { return q['Ref No']; }));
  const now = new Date();

  return filteredQueue.map(function (q) {
    const refNo = s_(q['Ref No']).toUpperCase();
    const p = propsByRef[refNo];
    const daysLive = (p && p.timestamp) ? Math.floor((now - new Date(p.timestamp)) / 86400000) : null;
    return {
      refNo: refNo,
      user: s_(q['User Name']),
      city: p ? p.city : '',
      propertyType: p ? p.propertyType : '',
      opportunityType: p ? p.opportunityType : '',
      currentStatus: p ? p.status : '',
      daysLive: daysLive,
      lastAction: s_(q['Status']),
      lastComment: s_(q['Action']),
      lastActionDate: isoDate_(q['Date'])
    };
  });
}

const REPUBLISH_QUEUE_SUPER_USER = 'Sherden';

function blockIfPendingRepublishItems_(actionVerb, currentUser) {

  const counts = getPendingRepublishCounts_();
  const users = Object.keys(counts);

  throw new Error('Cannot ' + actionVerb + ' the queue — these users still have pending items they haven\'t actioned yet: ' + summary + '. Ask them to finish reviewing their queue first, then try again.');
}

const SOCIAL_MEDIA_APPROVAL_ACTIONS = ['Publish', 'Republish', 'Drop', 'Lost', 'Hold', 'Closed', 'Data Change'];

function parseRequestedPlatforms_(raw) {
  raw = s_(raw);
  if (!raw) return SOCIAL_MEDIA_PLATFORMS.slice();
  const list = raw.split(',').map(function (s) { return s.trim(); }).filter(function (s) { return SOCIAL_MEDIA_PLATFORMS.indexOf(s) !== -1; });
  return list.length ? list : SOCIAL_MEDIA_PLATFORMS.slice();
}

function parseRequestedPlatformsRaw_(raw) {
  raw = s_(raw);
  if (!raw) return [];
  return raw.split(',').map(function (s) { return s.trim(); }).filter(function (s) { return SOCIAL_MEDIA_PLATFORMS.indexOf(s) !== -1; });
}

function getSocialMediaQueue(sessionToken, filters) {
  requireSocialMediaQueueAccess_(sessionToken);
  filters = filters || {};
  const rows = supabaseRequest_('GET', SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, ['select=*', 'order=id.desc']) || [];
  if (!rows.length) return [];
  let activeRows = rows.filter(function (q) { return !s_(q['Completed Date']); });

  // Filter by Approved Date range, if provided.
  if (filters.dateFrom || filters.dateTo) {
    const from = filters.dateFrom ? new Date(filters.dateFrom + 'T00:00:00.000Z') : null;
    const to = filters.dateTo ? new Date(filters.dateTo + 'T23:59:59.999Z') : null;
    activeRows = activeRows.filter(function (q) {
      const approvedIso = isoDate_(q['Approved Date']);
      if (!approvedIso) return false;
      const d = new Date(approvedIso);
      if (from && d < from) return false;
      if (to && d > to) return false;
      return true;
    });
  }

  // Filter by Approved Action (Publish, Republish, Drop, Lost, Hold, Closed, Data Change), if provided.
  if (filters.status && filters.status !== 'All') {
    activeRows = activeRows.filter(function (q) { return s_(q['Approved Action']) === filters.status; });
  }

  if (!activeRows.length) return [];
  const propsByRef = getPropertiesByRefNos_(activeRows.map(function (q) { return q['Ref No']; }));
  return activeRows.map(function (q) {
    const refNo = s_(q['Ref No']).toUpperCase();
    const p = propsByRef[refNo];
    const platforms = {};
    SOCIAL_MEDIA_PLATFORMS.forEach(function (plat) { platforms[plat] = isoDate_(q[plat + ' Date']); });
    return {
      refNo: refNo,
      approvedAction: s_(q['Approved Action']),
      approvedBy: s_(q['Approved By']),
      approvedDate: isoDate_(q['Approved Date']),
      city: p ? p.city : '',
      propertyType: p ? p.propertyType : '',
      opportunityType: p ? p.opportunityType : '',
      currentStatus: p ? p.status : '',
      platforms: platforms,
      comment: s_(q['Comment']),
      requestedPlatforms: parseRequestedPlatforms_(q['Requested Platforms'])
    };
  });
}

function getMyDashboardStats(sessionToken, dateFrom, dateTo) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  const currentUser = session.name;
  const from = dateFrom ? new Date(dateFrom + 'T00:00:00.000Z') : null;
  const to = dateTo ? new Date(dateTo + 'T23:59:59.999Z') : null;

  const propRows = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
    'User=eq.' + encodeURIComponent(currentUser), 'select=' + encodeURIComponent('Ref No,Timestamp'), 'limit=5000'
  ]) || [];
  const propertiesAdded = propRows.filter(function (r) {
    const ts = isoDate_(r['Timestamp']);
    if (!ts) return false;
    const d = new Date(ts);
    if (from && d < from) return false;
    if (to && d > to) return false;
    return true;
  }).length;

  const isTrackedAgent = INV_AGENTS.indexOf(currentUser) !== -1;
  let salesValue = 0;
  const currentYear = new Date().getFullYear();
  const quarterly = { Q1: 0, Q2: 0, Q3: 0, Q4: 0 };
  if (isTrackedAgent) {
    getAllInvoiceRows_().forEach(function (r) {
      if (r.agents[currentUser] === null || r.agents[currentUser] === undefined) return;
      if (!r.invDate) return;
      const amt = r.agents[currentUser] || 0;
      if (r.invDate.getFullYear() === currentYear) {
        const q = Math.floor(r.invDate.getMonth() / 3) + 1;
        quarterly['Q' + q] += amt;
      }
      if (from && r.invDate < from) return;
      if (to && r.invDate > to) return;
      salesValue += amt;
    });
  }

  let target = null, achievedThisYear = 0, remainingTarget = null, achievedPct = null, paceNeeded = null, monthsRemaining = null;
  if (isTrackedAgent) {
    achievedThisYear = quarterly.Q1 + quarterly.Q2 + quarterly.Q3 + quarterly.Q4;
    const targetsForYear = INV_TARGETS_BY_YEAR[String(currentYear)];
    if (targetsForYear && targetsForYear[currentUser] !== undefined) {
      target = targetsForYear[currentUser];
      remainingTarget = Math.max(target - achievedThisYear, 0);
      achievedPct = target ? Math.min(100, Math.round(achievedThisYear / target * 100)) : 0;
      const now = new Date();
      monthsRemaining = 12 - now.getMonth();
      paceNeeded = monthsRemaining > 0 ? Math.round(remainingTarget / monthsRemaining) : 0;
    }
  }

  // Admins see the company-wide C7 brokering total instead of
  // (necessarily small or zero) personal sales — summed from c7Booking
  // specifically, which is C7's own brokering revenue on each invoice,
  // not the full invoiceAmt (which can include amounts that aren't
  // actually C7's, like owner proceeds or other parties' shares).
  // Deliberately NOT gated by the selected from/to filter — this is
  // always the full current calendar year, so switching the date
  // filter to e.g. just this month never makes this figure look like
  // "LKR 0"; that filter still applies to the Admin Overview cards
  // below it, just not to this headline number. Car Park revenue is
  // excluded from both this total and the quarterly breakdown (so the
  // total always equals the sum of the four quarters shown).
  let companySalesValue = null;
  const companyQuarterly = { Q1: 0, Q2: 0, Q3: 0, Q4: 0 };
  if (session.role === 'Admin') {
    companySalesValue = 0;
    getAllInvoiceRows_().forEach(function (r) {
      if (!r.invDate) return;
      if (r.invoiceFor === 'Car Park') return;
      if (r.invDate.getFullYear() !== currentYear) return;
      const q = Math.floor(r.invDate.getMonth() / 3) + 1;
      companyQuarterly['Q' + q] += r.c7Booking;
      companySalesValue += r.c7Booking;
    });
  }

  return {
    propertiesAdded: propertiesAdded, salesValue: salesValue, isTrackedAgent: isTrackedAgent,
    isAdmin: session.role === 'Admin', companySalesValue: companySalesValue, companyQuarterly: companyQuarterly,
    quarterly: quarterly, quarterlyYear: currentYear,
    target: target, achievedThisYear: achievedThisYear, remainingTarget: remainingTarget,
    achievedPct: achievedPct, paceNeeded: paceNeeded, monthsRemaining: monthsRemaining
  };
}

/* Admin-only operational overview for the given date range: how many
   properties were added, how many the social media person actually
   finished publishing, how many Ikman/Facebook boost requests are still
   outstanding across the whole team (informational counts only — the
   actionable, per-admin-filtered worklist is still Activity Log), and
   how many properties were Dropped/Lost/Closed/Republished in that
   window. Uses small, targeted queries (not getAllDataRows_) so this
   stays fast regardless of how large the properties table grows. */
function getAdminDashboardOverview(sessionToken, dateFrom, dateTo) {
  requireAdminOverviewAccess_(sessionToken);
  const fromIso = dateFrom ? (dateFrom + 'T00:00:00.000Z') : null;
  const toIso = dateTo ? (dateTo + 'T23:59:59.999Z') : null;

  const tsCol = encodeURIComponent('Timestamp');
  const propsQuery = ['select=' + encodeURIComponent('Ref No,User')];
  if (fromIso) propsQuery.push(tsCol + '=gte.' + encodeURIComponent(fromIso));
  if (toIso) propsQuery.push(tsCol + '=lte.' + encodeURIComponent(toIso));

  const cdCol = encodeURIComponent('Completed Date');
  const smQuery = ['select=' + encodeURIComponent('Ref No')];
  if (fromIso) smQuery.push(cdCol + '=gte.' + encodeURIComponent(fromIso));
  if (toIso) smQuery.push(cdCol + '=lte.' + encodeURIComponent(toIso));

  const dtCol = encodeURIComponent('"Date & Time"');
  const suQuery = ['select=' + encodeURIComponent('Status'), 'Status=in.(Drop,Lost,Closed,Republish)'];
  if (fromIso) suQuery.push(dtCol + '=gte.' + encodeURIComponent(fromIso));
  if (toIso) suQuery.push(dtCol + '=lte.' + encodeURIComponent(toIso));

  const boostQuery = ['select=' + encodeURIComponent('Requested Platforms,Boost Completed'), 'Status=eq.' + encodeURIComponent('Boost Request')];

  // "Publish/Republish To Do" tracks the SOCIAL MEDIA team's own
  // backlog — items already approved by an Admin (so they're sitting in
  // the Social Media Queue) but not yet marked complete on all
  // requested platforms. This is a different stage from admin approval:
  // an item leaves the Activity Log the moment it's approved, then sits
  // here until someone actually publishes it on Ikman/LPW/Facebook/etc.
  // Deliberately not gated by the selected date range, same reasoning
  // as the other backlog-style figures on this page — an unfinished
  // item from last week is still unfinished today.
  const smWorkQuery = ['select=' + encodeURIComponent('Approved Action,Completed Date'), 'Approved Action=in.(Publish,Republish)'];

  const results = supabaseRequestsParallel_([
    { method: 'GET', table: SUPABASE_PROPERTIES_TABLE, query: propsQuery },
    { method: 'GET', table: SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, query: smQuery },
    { method: 'GET', table: SUPABASE_STATUS_UPDATE_TABLE, query: suQuery },
    { method: 'GET', table: SUPABASE_STATUS_UPDATE_TABLE, query: boostQuery },
    { method: 'GET', table: SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, query: smWorkQuery }
  ]);

  const propsRows = results[0] || [];
  const smRows = results[1] || [];
  const suRows = results[2] || [];
  const boostRows = results[3] || [];
  const smWorkRows = results[4] || [];

  const statusCounts = { Drop: 0, Lost: 0, Closed: 0, Republish: 0 };
  suRows.forEach(function (r) {
    const st = s_(r['Status']);
    if (statusCounts.hasOwnProperty(st)) statusCounts[st]++;
  });

  let pendingIkman = 0, pendingFacebook = 0;
  boostRows.forEach(function (r) {
    if (s_(r['Boost Completed'])) return;
    const plat = s_(r['Requested Platforms']);
    if (plat === 'Ikman') pendingIkman++;
    else if (plat === 'Facebook') pendingFacebook++;
  });

  // Group Properties Added (within the selected date range) by agent,
  // for the "published by agent this month" breakdown on My Dashboard.
  const publishedByAgentMap = {};
  propsRows.forEach(function (r) {
    const agent = s_(r['User']) || '(Unassigned)';
    publishedByAgentMap[agent] = (publishedByAgentMap[agent] || 0) + 1;
  });
  const publishedByAgent = Object.keys(publishedByAgentMap)
    .map(function (agent) { return { agent: agent, count: publishedByAgentMap[agent] }; })
    .sort(function (a, b) { return b.count - a.count; });

  let pendingPublishCount = 0, pendingRepublishCount = 0;
  smWorkRows.forEach(function (r) {
    if (s_(r['Completed Date'])) return; // already finished — not part of the backlog
    const action = s_(r['Approved Action']);
    if (action === 'Publish') pendingPublishCount++;
    else if (action === 'Republish') pendingRepublishCount++;
  });

  return {
    propertiesAdded: propsRows.length,
    propertiesPublishedBySocialMedia: smRows.length,
    droppedCount: statusCounts.Drop,
    lostCount: statusCounts.Lost,
    closedCount: statusCounts.Closed,
    republishedCount: statusCounts.Republish,
    pendingIkmanBumpUps: pendingIkman,
    pendingFacebookBoosts: pendingFacebook,
    publishedByAgent: publishedByAgent,
    pendingPublishCount: pendingPublishCount,
    pendingRepublishCount: pendingRepublishCount
  };
}

      'select=' + encodeURIComponent('Requested Platforms,Completed Date'), 'limit=1'
    ]);
    const priorRow = existing && existing[0];
    validPlatforms = (priorRow && s_(priorRow['Completed Date']))
      ? parseRequestedPlatforms_(priorRow['Requested Platforms'])
      : SOCIAL_MEDIA_PLATFORMS.slice();
  } else {
    validPlatforms = (platforms || []).filter(function (p) { return SOCIAL_MEDIA_PLATFORMS.indexOf(p) !== -1; });
    if (!validPlatforms.length) {
      throw new Error('Choose at least one platform to publish on.');
    }
  }
  const now = new Date();
  const row = {
    'Ref No': refNo, 'Approved Action': status, 'Approved By': currentUser, 'Approved Date': now,
    'Requested Platforms': validPlatforms.join(', '),
    'Comment': s_(comment),
    'Completed Date': null
  };
  SOCIAL_MEDIA_PLATFORMS.forEach(function (plat) { row[plat + ' Date'] = null; });

  supabaseRequestsParallel_([
    {
      method: 'POST', table: SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE,
      query: ['on_conflict=' + encodeURIComponent('Ref No')],
      body: [row], extraHeaders: { 'Prefer': 'resolution=merge-duplicates,return=minimal' }
    },
    {
      method: 'POST', table: SUPABASE_STATUS_UPDATE_TABLE, query: null,
      body: [{
        'Date & Time': now, 'Ref No': refNo, 'User': currentUser,
        'Status': 'Social Media Requested',
        'Comment': 'Requested on: ' + validPlatforms.join(', ')
      }],
      extraHeaders: { 'Prefer': 'return=minimal' }
    }
  ]);

  return { success: true };
}

const REMOVAL_ACTIONS = ['Drop', 'Lost', 'Hold', 'Closed'];

/* Pushes a property into the Social Media Queue as a removal task whenever
   it's marked Drop/Lost/Hold/Closed, so whoever manages the listings on
   Ikman/LPW/Facebook/Instagram knows to take the ad down. Shared by
   both updatePropertyStatus_internal_ (status changes from My Properties)
   and submitRepublishQueueAction (status changes from the Republish Queue)
   so marking something Lost/Drop/Hold/Closed always triggers this,
   regardless of which screen the change was made from. */
function pushRemovalToSocialMediaQueue_(refNo, newStatus, currentUser) {
  if (REMOVAL_ACTIONS.indexOf(newStatus) === -1) return;
  const now = new Date();

  // Carry forward the platforms this property was actually requested/
  // published on previously, so a removal only flags the platforms the
  // ad is actually live on (e.g. only Ikman & LPW, not Facebook and
  // Instagram too if it was never posted there). Falls back to blank
  // only if there's no prior queue row at all — parseRequestedPlatforms_
  // already treats a blank value as "all platforms" downstream, which is
  // a reasonable default only for a property with no publishing history.
  const existing = supabaseRequest_('GET', SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, [
    encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo),
    'select=' + encodeURIComponent('Requested Platforms'), 'limit=1'
  ]);
  const priorRow = existing && existing[0];
  const requestedPlatforms = priorRow ? s_(priorRow['Requested Platforms']) : '';

  // Comment is still explicitly cleared (not carried forward) so a
  // removal action never inherits a stale note from an earlier, unrelated
  // approval on this same property (e.g. a past price-update Publish).
  const queueRow = {
    'Ref No': refNo, 'Approved Action': newStatus, 'Approved By': currentUser || '',
    'Approved Date': now, 'Completed Date': null,
    'Comment': '', 'Requested Platforms': requestedPlatforms
  };
  SOCIAL_MEDIA_PLATFORMS.forEach(function (plat) { queueRow[plat + ' Date'] = null; });
  supabaseRequest_('POST', SUPABASE_SOCIAL_MEDIA_QUEUE_TABLE, ['on_conflict=' + encodeURIComponent('Ref No')],
    [queueRow], { 'Prefer': 'resolution=merge-duplicates,return=minimal' });
}

function updatePropertyStatus_internal_(refNo, newStatus, currentUser, comment) {
  if (STATUS_CHANGE_OPTIONS.indexOf(newStatus) === -1) {
    throw new Error('"' + newStatus + '" is not a valid status change option.');
  }
  refNo = s_(refNo).toUpperCase();

  const results = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
    encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo), 'select=' + encodeURIComponent('Ref No,User,Status'), 'limit=1'
  ]);
  if (!results || !results.length) throw new Error('Property ' + refNo + ' not found.');
  const ownerName = s_(results[0]['User']);
  const currentStatus = s_(results[0]['Status']) || 'Active';

  if (ownerName.toLowerCase() !== s_(currentUser).toLowerCase() && getUserRole_(currentUser) !== 'Admin') {
    throw new Error('Only ' + ownerName + ' (or an Admin) can change this property\'s status.');
  }

  const REACTIVATING_ACTIONS = ['Publish', 'Republish', 'New Ad Published'];
  let sheetStatus;
  if (newStatus === 'Data Change') {
    sheetStatus = currentStatus;
  } else {
    sheetStatus = REACTIVATING_ACTIONS.indexOf(newStatus) !== -1 ? 'Active' : newStatus;
    updatePropertyField_(refNo, 'Status', sheetStatus);
  }

  // CHANGED: Republish and Data Change entries created here (status
  // change from My Properties) are routed to Keerthie only, same as
  // new-listing Publish entries and Republish from the Republish Queue.
  const logEntry = {
    'Date & Time': new Date(), 'Ref No': refNo, 'User': currentUser || '',
    'Status': newStatus, 'Comment': comment || ''
  };
  if (newStatus === 'Republish' || newStatus === 'Data Change') logEntry['Assigned To'] = ACTIVITY_LOG_NEW_LISTING_ADMIN;
  supabaseRequest_('POST', SUPABASE_STATUS_UPDATE_TABLE, null, [logEntry], { 'Prefer': 'return=minimal' });

  pushRemovalToSocialMediaQueue_(refNo, newStatus, currentUser);

  return { success: true, refNo: refNo, status: sheetStatus };
}

/* Admin-only, drives the data change directly instead of via Supabase UI. */
function adminUpdatePropertyFields(sessionToken, refNo, fields) {
  const currentUser = requireAdmin_(sessionToken);
  refNo = s_(refNo).toUpperCase();
  fields = fields || {};

  let propertyType = s_(fields.propertyType);
  if (!propertyType) {
    const existing = supabaseRequest_('GET', SUPABASE_PROPERTIES_TABLE, [
      encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo), 'select=' + encodeURIComponent('Property Type'), 'limit=1'
    ]);
    if (!existing || !existing.length) throw new Error('Property ' + refNo + ' not found.');
    propertyType = s_(existing[0]['Property Type']);
  }

  const body = {};
  const changes = [];
  function setIfProvided(key, colName, label) {
    if (!colName) return;
    const val = fields[key];
    if (val === undefined || val === null || val === '') return;
    body[colName] = val;
    changes.push(label + ': ' + val);
  }

  setIfProvided('opportunityType', 'Opportunity Type', 'Opportunity Type');
  setIfProvided('propertyType', 'Property Type', 'Property Type');
  setIfProvided('address', 'Address', 'Address');
  setIfProvided('city', 'City', 'City');
  setIfProvided('priceTotal', 'Price Total', 'Price Total');
  setIfProvided('pricePerPerch', 'Price per Perch', 'Price per Perch');
  setIfProvided('furnished', 'Furnished', 'Furnished');
  setIfProvided('contact1', 'Contact No 1', 'Contact No 1');
  setIfProvided('user', 'User', 'Agent');
  setIfProvided('apartmentComplex', 'Apartment Complex', 'Apartment Complex');
  setIfProvided('bedRooms', bedRoomsColumnFor_(propertyType), 'Bed Rooms');
  setIfProvided('bathRooms', bathRoomsColumnFor_(propertyType), 'Bath Rooms');
  setIfProvided('floorArea', floorAreaColumnFor_(propertyType), 'Floor Area');
  setIfProvided('landSize', landSizeColumnFor_(propertyType), 'Land Size');
  setIfProvided('floor', 'Floor', 'Floor');

  // "Other Information" as shown to the user is actually assembled by
  // joining up to 5 different source columns depending on property type
  // (see rowToObject_'s otherInfo logic) — House/Land/Apartment/
  // Commercial each historically wrote to a different one, plus the
  // generic "Comments" column used across all types. To avoid an edit
  // here creating duplicate/stale text (the old column's text still
  // showing alongside the newly-edited text), an edit consolidates
  // everything into "Comments" going forward and clears the other four,
  // so there's a single, unambiguous source of truth after any edit.
  if (fields.comments !== undefined) {
    body['Comments'] = fields.comments;
    body['Other information'] = null;
    body['Other information_1'] = null;
    body['Comments_1'] = null;
    body['Comments_2'] = null;
    changes.push('Other Information updated');
  }

  if (!Object.keys(body).length) return { success: true, refNo: refNo, changed: false };

  supabaseRequest_('PATCH', SUPABASE_PROPERTIES_TABLE, [encodeURIComponent('Ref No') + '=eq.' + encodeURIComponent(refNo)], body, { 'Prefer': 'return=minimal' });

  supabaseRequest_('POST', SUPABASE_STATUS_UPDATE_TABLE, null,
    [{ 'Date & Time': new Date(), 'Ref No': refNo, 'User': currentUser, 'Status': 'Admin Edit', 'Comment': changes.join('; ') }],
    { 'Prefer': 'return=minimal' });

  return { success: true, refNo: refNo, changed: true };
}

/* ============================== BULK UPLOAD ============================== */

function bulkImportRows(sessionToken, pastedText, dryRun) {
  const currentUser = requireAdmin_(sessionToken);
  if (!pastedText || !pastedText.trim()) throw new Error('No rows to import.');
  const lines = pastedText.split(/\r?\n/).filter(function (l) { return l.trim().length > 0; });
  let dataLines = lines;
  if (lines.length && lines[0].toLowerCase().indexOf('ref no') !== -1 && lines[0].toLowerCase().indexOf('timestamp') !== -1) {
    dataLines = lines.slice(1);
  }

  const existing = getAllDataRows_();
  const refToRow = {};
  existing.forEach(function (r) { refToRow[s_(r.cells[COL.REF_NO - 1]).toUpperCase()] = r.rowIndex; });

  let updated = 0, added = 0, skipped = 0;
  const allRows = [];
  const logEntries = [];

  dataLines.forEach(function (line) {
    const cells = line.split('\t');
    if (!cells[0] || !s_(cells[0])) { skipped++; return; }
    const refNo = s_(cells[0]).toUpperCase();
    const padded = new Array(TOTAL_COLS).fill('');
    for (let i = 0; i < TOTAL_COLS; i++) padded[i] = i < cells.length ? cells[i] : '';
    padded[0] = refNo;
    if (!padded[COL.TIMESTAMP - 1]) padded[COL.TIMESTAMP - 1] = new Date().toISOString();
    padded[COL.CONTACT1 - 1] = s_(padded[COL.CONTACT1 - 1]);
    padded[COL.CONTACT2 - 1] = s_(padded[COL.CONTACT2 - 1]);

    allRows.push(padded);
    const isUpdate = !!refToRow[refNo];
    if (isUpdate) updated++; else added++;
    logEntries.push({
      'Date & Time': new Date(), 'Ref No': refNo, 'User': currentUser,
      'Status': isUpdate ? 'Data Change' : 'Publish', 'Comment': 'Bulk import'
    });
  });

  if (!dryRun && allRows.length) {
    upsertPropertyRows_(allRows);
    supabaseRequest_('POST', SUPABASE_STATUS_UPDATE_TABLE, null, logEntries, { 'Prefer': 'return=minimal' });
  }

  return { updated: updated, added: added, skipped: skipped, dryRun: !!dryRun };
}

function bulkImportFile(sessionToken, base64Data, filename, dryRun) {
  requireAdmin_(sessionToken);
  const bytes = Utilities.base64Decode(base64Data);
  const blob = Utilities.newBlob(bytes, MimeType.MICROSOFT_EXCEL, filename || 'import.xlsx');

  let tempFile;
  try {
    tempFile = Drive.Files.create(
      { name: 'TEMP_IMPORT_' + new Date().getTime(), mimeType: MimeType.GOOGLE_SHEETS },
      blob,
      { convert: true }
    );
  } catch (err) {
    throw new Error('Could not convert the uploaded file. Make sure the "Drive API" advanced service is enabled in this Apps Script project. Original error: ' + err.message);
  }

  try {
    const tempSs = SpreadsheetApp.openById(tempFile.id);
    let sheet = tempSs.getSheetByName(SHEET_NAME) || tempSs.getSheets()[0];
    const values = sheet.getDataRange().getValues();
    if (!values.length) return { updated: 0, added: 0, skipped: 0, dryRun: !!dryRun };

    const tsv = values.map(function (row) { return row.join('\t'); }).join('\n');
    return bulkImportRows(sessionToken, tsv, dryRun);
  } finally {
    Drive.Files.remove(tempFile.id);
  }
}

function n0_(v) {
  if (v === null || v === undefined || v === '') return 0;
  const num = typeof v === 'number' ? v : parseFloat(String(v).replace(/[^0-9.\-]/g, ''));
  return isNaN(num) ? 0 : num;
}

function classifyInvoice_(details) {
  const d = (details || '').toLowerCase();
  if (d.indexOf('sale') !== -1) return 'Sale';
  if (d.indexOf('rent') !== -1) return 'Rental';
  return 'Other';
}

const HEADLINE_CATEGORIES = ['Sale', 'Rental', 'Management Fee', 'Car Park', 'Others'];
function classifyHeadline_(category, invoiceFor) {
  if (category === 'Sale' || category === 'Rental') return category;
  if (invoiceFor === 'Car Park') return 'Car Park';
  if (invoiceFor === 'C7 Management') return 'Management Fee';
  return 'Others';
}

let _invRowsCache = null;
let _invHeaderMapCache = {};
function getInvoiceHeaderMap_(sh) {
  if (_invHeaderMapCache[sh.getSheetId()]) return _invHeaderMapCache[sh.getSheetId()];
  const result = getUsersHeaderMap_(sh);
  _invHeaderMapCache[sh.getSheetId()] = result;
  return result;
}

function getAllInvoiceRows_() {
  if (_invRowsCache) return _invRowsCache;
  const ss = SpreadsheetApp.openById(INVOICE_SPREADSHEET_ID);
  const rows = [];
  INV_YEAR_SHEETS.forEach(function (yearName) {
    const sh = ss.getSheetByName(yearName);
    if (!sh) return;
    const lastRow = sh.getLastRow();
    if (lastRow < 2) return;
    const { map, lastCol } = getInvoiceHeaderMap_(sh);
    const detailsCol = map['details'];
    const invoiceNoCol = map['invoiceno'];
    const customerCol = map['customername'];
    const invDateCol = map['invdate'] || map['duedate'];
    const invoiceAmtCol = map['invoiceamt'];
    const paymentAmtCol = map['paymentamt'];
    const toBeReceivedCol = map['tobereceived'];
    const invoiceForCol = map['incoicefor'] || map['invoicefor'];
    const c7BookingCol = map['cbooking'] || map['c7booking'];
    if (!invoiceNoCol || !detailsCol) return;
    const agentTextCol = map['agent'];

    const values = sh.getRange(2, 1, lastRow - 1, lastCol).getValues();
    values.forEach(function (r) {
      const invoiceNo = s_(r[invoiceNoCol - 1]);
      if (!invoiceNo) return;
      const invDate = invDateCol ? r[invDateCol - 1] : null;
      const agents = {};
      const invAmtForRow = invoiceAmtCol ? n0_(r[invoiceAmtCol - 1]) : 0;
      INV_AGENTS.forEach(function (a) {
        const col = map[a.toLowerCase()];
        if (col) {
          agents[a] = n0_(r[col - 1]) || null;
        } else if (agentTextCol) {
          const agentText = s_(r[agentTextCol - 1]).toLowerCase();
          agents[a] = agentText.indexOf(a.toLowerCase()) !== -1 ? invAmtForRow : null;
        } else {
          agents[a] = null;
        }
      });
      rows.push({
        year: yearName,
        invDate: invDate instanceof Date ? invDate : null,
        details: s_(r[detailsCol - 1]),
        customerName: customerCol ? s_(r[customerCol - 1]) : '',
        invoiceNo: invoiceNo,
        invoiceAmt: invoiceAmtCol ? n0_(r[invoiceAmtCol - 1]) : 0,
        paymentAmt: paymentAmtCol ? n0_(r[paymentAmtCol - 1]) : 0,
        toBeReceived: toBeReceivedCol ? n0_(r[toBeReceivedCol - 1]) : 0,
        invoiceFor: invoiceForCol ? s_(r[invoiceForCol - 1]) : '',
        c7Booking: c7BookingCol ? n0_(r[c7BookingCol - 1]) : 0,
        agents: agents,
        category: classifyInvoice_(r[detailsCol - 1])
      });
    });
  });
  _invRowsCache = rows;
  return rows;
}

function buildInvoiceTargets_(year, byAgent) {
  const targets = INV_TARGETS_BY_YEAR[String(year)];
  if (!targets) return null;
  function block(agentName) {
    const achieved = (byAgent.find(function (a) { return a.agent === agentName; }) || { total: 0 }).total;
    const target = targets[agentName];
    return { target: target, remaining: Math.max(target - achieved, 0), achievedPct: target ? (achieved / target * 100) : 0 };
  }
  return { Theeban: block('Theeban'), Chaminda: block('Chaminda'), Jeyandran: block('Jeyandran') };
}

function invInQuarter_(month, quarter) {
  if (!quarter || quarter === 'All') return true;
  const q = parseInt(quarter, 10);
  return Math.ceil(month / 3) === q;
}

function getQuarterlyAgentPerformance(sessionToken, filters) {
  requireAdmin_(sessionToken);
  filters = filters || {};
  const rows = getAllInvoiceRows_();

  const useDateRange = filters.dateFrom && filters.dateTo;
  const rangeFrom = useDateRange ? new Date(filters.dateFrom) : null;
  const rangeTo = useDateRange ? new Date(filters.dateTo + 'T23:59:59') : null;

  const filtered = rows.filter(function (r) {
    if (!r.invDate) return false;
    if (useDateRange) {
      if (r.invDate < rangeFrom || r.invDate > rangeTo) return false;
    } else {
      if (filters.year && filters.year !== 'All' && r.year !== String(filters.year)) return false;
      if (filters.month && filters.month !== 'All' && (r.invDate.getMonth() + 1) !== parseInt(filters.month, 10)) return false;
      if (filters.quarter && filters.quarter !== 'All' && !invInQuarter_(r.invDate.getMonth() + 1, filters.quarter)) return false;
    }
    return true;
  });

  const agents = INV_AGENTS.filter(function (a) { return a !== 'Keerthie' && a !== 'Sherden'; });
  const quarterMap = {};

  filtered.forEach(function (r) {
    const q = Math.ceil((r.invDate.getMonth() + 1) / 3);
    const key = r.year + '-Q' + q;
    if (!quarterMap[key]) quarterMap[key] = {};
    agents.forEach(function (a) {
      const amt = r.agents[a] || 0;
      quarterMap[key][a] = (quarterMap[key][a] || 0) + amt;
    });
  });

  const quarters = Object.keys(quarterMap).sort();
  const data = quarters.map(function (k) {
    const row = { quarter: k };
    agents.forEach(function (a) { row[a] = quarterMap[k][a] || 0; });
    return row;
  });

  return { agents: agents, quarters: quarters, data: data };
}

function getInvoiceDashboardData(sessionToken, filters) {
  requireAdmin_(sessionToken);
  filters = filters || {};
  const rows = getAllInvoiceRows_();

  const useDateRange = filters.dateFrom && filters.dateTo;
  const rangeFrom = useDateRange ? new Date(filters.dateFrom) : null;
  const rangeTo = useDateRange ? new Date(filters.dateTo + 'T23:59:59') : null;
  const selectedUsers = (filters.users && filters.users.length) ? filters.users : null;

  const matching = rows.filter(function (r) {
    if (!r.invDate) return false;
    if (useDateRange) {
      if (r.invDate < rangeFrom || r.invDate > rangeTo) return false;
    } else {
      if (filters.year && filters.year !== 'All' && r.year !== String(filters.year)) return false;
      if (filters.month && filters.month !== 'All' && (r.invDate.getMonth() + 1) !== parseInt(filters.month, 10)) return false;
      if (filters.quarter && filters.quarter !== 'All' && !invInQuarter_(r.invDate.getMonth() + 1, filters.quarter)) return false;
    }
    if (selectedUsers) {
      const hasAny = selectedUsers.some(function (u) { return r.agents[u] !== null && r.agents[u] !== undefined; });
      if (!hasAny) return false;
    }
    return true;
  });

  function amountFor(r) {
    if (selectedUsers) {
      return selectedUsers.reduce(function (sum, u) { return sum + (r.agents[u] || 0); }, 0);
    }
    return r.invoiceAmt;
  }

  function summarize(list) {
    const summary = { Sale: { count: 0, amount: 0 }, Rental: { count: 0, amount: 0 }, Other: { count: 0, amount: 0 } };
    const headline = {};
    HEADLINE_CATEGORIES.forEach(function (k) { headline[k] = { count: 0, amount: 0 }; });
    let totalInvoiced = 0, totalReceived = 0, totalOutstanding = 0;
    list.forEach(function (r) {
      const amt = amountFor(r);
      summary[r.category].count++;
      summary[r.category].amount += amt;
      const hKey = classifyHeadline_(r.category, r.invoiceFor);
      headline[hKey].count++;
      headline[hKey].amount += amt;
      totalInvoiced += r.invoiceAmt;
      totalReceived += r.paymentAmt;
      totalOutstanding += r.toBeReceived;
    });
    return {
      sale: summary.Sale, rental: summary.Rental, other: summary.Other,
      headline: headline,
      totalCount: list.length, totalInvoiced: totalInvoiced,
      totalReceived: totalReceived, totalOutstanding: totalOutstanding
    };
  }

  const overall = summarize(matching);

  const byAgent = INV_AGENTS.map(function (agent) {
    const agentRows = matching.filter(function (r) { return r.agents[agent] !== null && r.agents[agent] !== undefined; });
    const sale = agentRows.filter(function (r) { return r.category === 'Sale'; }).reduce(function (s, r) { return s + (r.agents[agent] || 0); }, 0);
    const rental = agentRows.filter(function (r) { return r.category === 'Rental'; }).reduce(function (s, r) { return s + (r.agents[agent] || 0); }, 0);
    const other = agentRows.filter(function (r) { return r.category === 'Other'; }).reduce(function (s, r) { return s + (r.agents[agent] || 0); }, 0);
    return { agent: agent, count: agentRows.length, sale: sale, rental: rental, other: other, total: sale + rental + other };
  }).filter(function (a) { return a.count > 0; });

  const byCategoryMap = {};
  matching.forEach(function (r) {
    if (!r.invoiceFor) return;
    byCategoryMap[r.invoiceFor] = (byCategoryMap[r.invoiceFor] || 0) + amountFor(r);
  });

  const allYearsNumeric = INV_YEAR_SHEETS.map(Number);
  const refYear = (filters.year && filters.year !== 'All') ? parseInt(filters.year, 10) : Math.max.apply(null, allYearsNumeric);
  const comparisonYears = [refYear, refYear - 1, refYear - 2, refYear - 3].filter(function (y) { return allYearsNumeric.indexOf(y) !== -1; });
  const categoryByYear = {};
  comparisonYears.forEach(function (y) {
    categoryByYear[y] = {};
    rows.forEach(function (r) {
      if (r.year !== String(y) || !r.invDate || !r.invoiceFor) return;
      if (selectedUsers && !selectedUsers.some(function (u) { return r.agents[u] !== null && r.agents[u] !== undefined; })) return;
      const amt = amountFor(r);
      categoryByYear[y][r.invoiceFor] = (categoryByYear[y][r.invoiceFor] || 0) + amt;
    });
  });

  const allCategoryKeys = {};
  Object.keys(byCategoryMap).forEach(function (k) { allCategoryKeys[k] = true; });
  comparisonYears.forEach(function (y) { Object.keys(categoryByYear[y]).forEach(function (k) { allCategoryKeys[k] = true; }); });

  const byCategory = Object.keys(allCategoryKeys).sort().map(function (k) {
    const years = {};
    comparisonYears.forEach(function (y) { years[y] = categoryByYear[y][k] || 0; });
    return { category: k, total: byCategoryMap[k] || 0, years: years };
  });
  const byCategoryGrandTotal = byCategory.reduce(function (s, c) { return s + c.total; }, 0);
  const byCategoryYearTotals = {};
  comparisonYears.forEach(function (y) { byCategoryYearTotals[y] = byCategory.reduce(function (s, c) { return s + c.years[y]; }, 0); });

  const outstandingInvoices = matching
    .filter(function (r) { return r.toBeReceived > 0; })
    .sort(function (a, b) { return a.invDate - b.invDate; })
    .map(function (r) {
      return {
        invoiceNo: r.invoiceNo, customerName: r.customerName,
        invDate: r.invDate.toISOString(), invoiceFor: r.invoiceFor,
        category: r.category, invoiceAmt: r.invoiceAmt, toBeReceived: r.toBeReceived
      };
    });

  const targets = (filters.year && filters.year !== 'All' && !useDateRange) ? buildInvoiceTargets_(filters.year, byAgent) : null;

  return {
    overall: overall,
    byAgent: byAgent,
    byCategory: byCategory,
    byCategoryGrandTotal: byCategoryGrandTotal,
    byCategoryYears: comparisonYears,
    byCategoryYearTotals: byCategoryYearTotals,
    outstandingInvoices: outstandingInvoices,
    targets: targets,
    matchedInvoices: matching.length
  };
}

const WIN_STATUSES_ = ['Closed'];
const LOSS_STATUSES_ = ['Lost', 'Drop', 'Duplicate', 'Obsolete'];

function listingAgentSort_(a, b) {
  const ia = USER_DISPLAY_ORDER.indexOf(a);
  const ib = USER_DISPLAY_ORDER.indexOf(b);
  const ra = ia === -1 ? USER_DISPLAY_ORDER.length : ia;
  const rb = ib === -1 ? USER_DISPLAY_ORDER.length : ib;
  if (ra !== rb) return ra - rb;
  return a.localeCompare(b);
}

function listingAgentsFromRows_(rows) {
  const set = {};
  rows.forEach(function (r) { if (r.user) set[r.user] = true; });
  return Object.keys(set).sort(listingAgentSort_);
}

let _listingRowsCache = null;
function getAllListingRows_() {
  if (_listingRowsCache) return _listingRowsCache;
  const rows = getAllDataRows_().map(rowToObject_).filter(function (r) { return !!r.timestamp; });
  _listingRowsCache = rows;
  return rows;
}

function getListingsFilterOptions(sessionToken) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  const isAdminUser = session.role === 'Admin';
  const rows = getAllListingRows_();
  const yearsSet = {};
  rows.forEach(function (r) { yearsSet[new Date(r.timestamp).getFullYear()] = true; });
  const years = Object.keys(yearsSet).sort(function (a, b) { return Number(a) - Number(b); });
  return {
    years: years,
    // Non-admins only ever see their own name here — not because the
    // dropdown itself enforces anything (a client could tamper with the

    // below actually allows. The real restriction is enforced there,
    // server-side, regardless of what gets sent from the browser.
    agents: isAdminUser ? listingAgentsFromRows_(rows) : [session.name],
    propertyTypes: PROPERTY_TYPES,
    statusList: STATUS_LIST,
    isAdmin: isAdminUser
  };
}

function getListingsDashboardData(sessionToken, filters) {
  const session = resolveSession_(sessionToken);
  touchSession_(sessionToken, session);
  filters = filters || {};
  // Non-admins can only ever see their own entries — this overrides
  // whatever "agent" value the client sent, so a regular user can never
  // view another agent's data by tampering with the request client-side.
  // Admins keep full access to any agent, or "All".
  if (session.role !== 'Admin') {
    filters = Object.assign({}, filters, { agent: session.name });
  }
  const allRows = getAllListingRows_();

  function matchesBase(r) {
    if (filters.propertyType && filters.propertyType !== 'All' && r.propertyType !== filters.propertyType) return false;
    if (filters.agent && filters.agent !== 'All' && r.user !== filters.agent) return false;
    return true;
  }
  function matchesTime(r) {
    const d = new Date(r.timestamp);
    if (filters.year && filters.year !== 'All' && d.getFullYear() !== parseInt(filters.year, 10)) return false;
    if (filters.month && filters.month !== 'All' && (d.getMonth() + 1) !== parseInt(filters.month, 10)) return false;
    return true;
  }

  const baseFiltered = allRows.filter(matchesBase);
  const filtered = baseFiltered.filter(matchesTime);

  const useYearlyTrend = !filters.year || filters.year === 'All';
  const trendSource = baseFiltered.filter(function (r) {
    if (useYearlyTrend) return true;
    return new Date(r.timestamp).getFullYear() === parseInt(filters.year, 10);
  });
  const trendAgents = (filters.agent && filters.agent !== 'All') ? [filters.agent] : listingAgentsFromRows_(trendSource);
  const trendMap = {};
  trendSource.forEach(function (r) {
    const d = new Date(r.timestamp);
    const key = useYearlyTrend ? String(d.getFullYear()) : (d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0'));
    if (!trendMap[key]) trendMap[key] = {};
    trendMap[key][r.user] = (trendMap[key][r.user] || 0) + 1;
  });
  const trendKeys = Object.keys(trendMap).sort();
  const trend = trendKeys.map(function (k) {
    const row = { label: k };
    trendAgents.forEach(function (a) { row[a] = trendMap[k][a] || 0; });
    return row;
  });

  const now = new Date();
  const curYear = now.getFullYear(), curMonth = now.getMonth();
  const dailySource = baseFiltered.filter(function (r) {
    const d = new Date(r.timestamp);
    return d.getFullYear() === curYear && d.getMonth() === curMonth;
  });
  const dailyAgents = (filters.agent && filters.agent !== 'All') ? [filters.agent] : listingAgentsFromRows_(dailySource);
  const dailyMap = {};
  dailySource.forEach(function (r) {
    const key = String(new Date(r.timestamp).getDate()).padStart(2, '0');
    if (!dailyMap[key]) dailyMap[key] = {};
    dailyMap[key][r.user] = (dailyMap[key][r.user] || 0) + 1;
  });
  const dailyKeys = Object.keys(dailyMap).sort();
  const daily = dailyKeys.map(function (k) {
    const row = { label: k };
    dailyAgents.forEach(function (a) { row[a] = dailyMap[k][a] || 0; });
    return row;
  });

  const statusByAgentMap = {};
  filtered.forEach(function (r) {
    const st = r.status || 'Active';
    if (!statusByAgentMap[r.user]) statusByAgentMap[r.user] = {};
    statusByAgentMap[r.user][st] = (statusByAgentMap[r.user][st] || 0) + 1;
  });
  const statusByAgentAgents = (filters.agent && filters.agent !== 'All') ? [filters.agent] : listingAgentsFromRows_(filtered);
  const statusByAgent = statusByAgentAgents.map(function (a) {
    const row = { agent: a };
    let total = 0;
    STATUS_LIST.forEach(function (s) { const n = (statusByAgentMap[a] && statusByAgentMap[a][s]) || 0; row[s] = n; total += n; });
    row.total = total;
    return row;
  });

  const statusByTypeMap = {};
  filtered.forEach(function (r) {
    const st = r.status || 'Active';
    const pt = r.propertyType || 'Unknown';
    if (!statusByTypeMap[pt]) statusByTypeMap[pt] = {};
    statusByTypeMap[pt][st] = (statusByTypeMap[pt][st] || 0) + 1;
  });
  const typesForStatus = (filters.propertyType && filters.propertyType !== 'All') ? [filters.propertyType] : PROPERTY_TYPES;
  const statusByType = typesForStatus.map(function (t) {
    const row = { type: t };
    let total = 0;
    STATUS_LIST.forEach(function (s) { const n = (statusByTypeMap[t] && statusByTypeMap[t][s]) || 0; row[s] = n; total += n; });
    row.total = total;
    return row;
  });

  const winLossAgents = (filters.agent && filters.agent !== 'All') ? [filters.agent] : listingAgentsFromRows_(filtered);
  const winLoss = winLossAgents.map(function (a) {
    const rows = filtered.filter(function (r) { return r.user === a; });
    const total = rows.length;
    const won = rows.filter(function (r) { return WIN_STATUSES_.indexOf(r.status) !== -1; }).length;
    const lost = rows.filter(function (r) { return LOSS_STATUSES_.indexOf(r.status) !== -1; }).length;
    const pending = total - won - lost;
    return {
      agent: a, total: total, won: won, lost: lost, pending: pending,
      winRate: total ? Math.round(won / total * 100) : 0,
      lossRate: total ? Math.round(lost / total * 100) : 0
    };
  });

  const cityMap = {};
  filtered.forEach(function (r) {
    const c = r.city || 'Unknown';
    cityMap[c] = (cityMap[c] || 0) + 1;
  });
  const cityBreakdown = Object.keys(cityMap)
    .map(function (c) { return { city: c, count: cityMap[c] }; })
    .sort(function (a, b) { return b.count - a.count; });

  return {
    matchedCount: filtered.length,
    trendAgents: trendAgents, trend: trend, trendIsYearly: useYearlyTrend,
    dailyAgents: dailyAgents, daily: daily,
    statusByAgent: statusByAgent,
    statusByType: statusByType,
    winLoss: winLoss,
    cityBreakdown: cityBreakdown
  };
}
